   149→export function TeamsView() {
   150→  const clubId = useAppStore((s) => s.currentClubId)
   151→  const role = useAppStore((s) => s.currentClub?.role)
   152→  const isExec = role === "executive"
   153→
   154→  const { data, isLoading, isError, refetch } = useQuery<TeamsResponse>({
   155→    queryKey: ["teams", clubId],
   156→    queryFn: () => api(`/api/clubs/${clubId}/teams`),
   157→    enabled: !!clubId,
   158→    // Realtime is primary; poll only as a fallback while the socket is down.
   159→    refetchInterval: usePollingFallback(30_000),
   160→  })
   161→
   162→  // Live presence — green dot on avatars of currently-online club members.
   163→  const online = usePresence(clubId ?? null)
   164→
   165→  const [selectedTeamId, setSelectedTeamId] = useState<string | null>(null)
   166→  const [createOpen, setCreateOpen] = useState(false)
   167→
   168→  const selectedTeam = data?.teams.find((t) => t.id === selectedTeamId) ?? null
   169→
   170→  if (!clubId) {
   171→    return <div className="p-8 text-sm text-muted-foreground">No club selected.</div>
   172→  }
   173→
   174→  return (
   175→    <div className="space-y-6">
   176→      <PageHeader
   177→        title="Teams"
   178→        description="Organize your club into working teams with dedicated rosters, tasks, and meetings."
   179→        actions={
   180→          isExec ? (
   181→            <Button variant="club" onClick={() => setCreateOpen(true)}>
   182→              <Plus className="mr-2 h-4 w-4" /> New team
   183→            </Button>
   184→          ) : undefined
   185→        }
   186→      />
   187→
   188→      {isError ? (
   189→        <div className="rounded-lg border border-red-200 bg-red-50 dark:border-red-900 dark:bg-red-950/40 p-4 text-sm text-red-700 dark:text-red-300">
   190→          Failed to load teams.{" "}
   191→          <button className="underline" onClick={() => refetch()}>
   192→            Try again
   193→          </button>
   194→        </div>
   195→      ) : isLoading ? (
   196→        <TeamsSkeleton />
   197→      ) : !data || data.teams.length === 0 ? (
   198→        <EmptyState
   199→          icon={<Users className="h-8 w-8" />}
   200→          title="No teams yet"
   201→          description={
   202→            isExec
   203→              ? "Create your first team to organize members around specific projects, events, or ongoing responsibilities."
   204→              : "Your club hasn't created any teams yet. Check back soon."
   205→          }
   206→          action={
   207→            isExec ? (
   208→              <Button variant="club" onClick={() => setCreateOpen(true)}>
   209→                <Plus className="mr-2 h-4 w-4" /> Create a team
   210→              </Button>
   211→            ) : undefined
   212→          }
   213→        />
   214→      ) : (
   215→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   216→          {data.teams.map((team) => (
   217→            <TeamCard
   218→              key={team.id}
   219→              clubId={clubId}
   220→              team={team}
   221→              isExec={isExec}
   222→              online={online}
   223→              onOpen={() => setSelectedTeamId(team.id)}
   224→            />
   225→          ))}
   226→        </div>
   227→      )}
   228→
   229→      {/* Team detail sheet */}
   230→      <Sheet open={!!selectedTeamId} onOpenChange={(o) => !o && setSelectedTeamId(null)}>
   231→        <SheetContent className="w-full sm:max-w-lg p-0 flex flex-col" side="right">
   232→          {selectedTeam ? (
   233→            <TeamDetailSheet
   234→              clubId={clubId}
   235→              team={selectedTeam}
   236→              isExec={isExec}
   237→              online={online}
   238→              onClose={() => setSelectedTeamId(null)}
   239→            />
   240→          ) : (
   241→            <div className="p-8 flex items-center justify-center">
   242→              <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   243→            </div>
   244→          )}
   245→        </SheetContent>
   246→      </Sheet>
   247→
   248→      <CreateTeamDialog clubId={clubId} open={createOpen} onOpenChange={setCreateOpen} />
   249→    </div>
   250→  )
   251→}

   271→function TeamCard({
   272→  clubId,
   273→  team,
   274→  isExec,
   275→  online,
   276→  onOpen,
   277→}: {
   278→  clubId: string
   279→  team: Team
   280→  isExec: boolean
   281→  online: Set<string>
   282→  onOpen: () => void
   283→}) {
   284→  const visibleMembers = team.members.slice(0, 5)
   285→  const overflow = team.members.length - visibleMembers.length
   286→  // Briefly highlight when this card was just touched by another user's
   287→  // realtime action (create/rename/member add/remove).
   288→  const flash = useRemoteChange("team", team.id)
   289→
   290→  return (
   291→    <div
   292→      role="button"
   293→      tabIndex={0}
   294→      aria-label={`Open team ${team.name}`}
   295→      onClick={onOpen}
   296→      onKeyDown={(e) => {
   297→        if (e.key === "Enter" || e.key === " ") {
   298→          e.preventDefault()
   299→          onOpen()
   300→        }
   301→      }}
   302→      className={cn(
   303→        "card-quiet p-5 cursor-pointer hover:shadow-md hover:border-club/40 transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-club animate-fade-in",
   304→        flash && "ring-2 ring-club/50 shadow-md"
   305→      )}
   306→    >
   307→      <div className="pb-3">
   308→        <div className="flex items-start justify-between gap-2">
   309→          <div className="min-w-0">
   310→            <h3 className="text-card-title truncate">{team.name}</h3>
   311→            {team.description ? (
   312→              <p className="text-caption mt-1 line-clamp-2">
   313→                {team.description}
   314→              </p>
   315→            ) : (
   316→              <p className="text-caption text-muted-foreground/70 mt-1 italic">No description</p>
   317→            )}
   318→          </div>
   319→          {isExec && <TeamCardMenu clubId={clubId} team={team} />}
   320→        </div>
   321→      </div>
   322→      <div className="space-y-3">
   323→        <div className="flex items-center gap-2 min-h-[32px]">
   324→          {team.members.length === 0 ? (
   325→            <span className="text-caption text-muted-foreground/70">No members yet</span>
   326→          ) : (
   327→            <div className="flex -space-x-2">
   328→              {visibleMembers.map((m) => (
   329→                <Avatar
   330→                  key={m.id}
   331→                  className="h-7 w-7 border-2 border-background"
   332→                  title={m.user.name}
   333→                >
   334→                  <AvatarImage src={m.user.avatarUrl ?? undefined} alt={m.user.name} />
   335→                  <AvatarFallback className="text-[10px]">{initials(m.user.name)}</AvatarFallback>
   336→                  {online.has(m.userId) && (
   337→                    <span
   338→                      aria-label="Online"
   339→                      className="absolute bottom-0 right-0 h-2 w-2 rounded-full bg-club ring-2 ring-background"
   340→                    />
   341→                  )}
   342→                </Avatar>
   343→              ))}
   344→              {overflow > 0 && (
   345→                <div className="h-7 w-7 rounded-full bg-muted border-2 border-background flex items-center justify-center text-[10px] font-medium text-muted-foreground">
   346→                  +{overflow}
   347→                </div>
   348→              )}
   349→            </div>
   350→          )}
   351→        </div>
   352→        <Separator />
   353→        <div className="flex items-center justify-between text-caption text-muted-foreground">
   354→          <span className="inline-flex items-center gap-1.5">
   355→            <Users className="h-3.5 w-3.5" />
   356→            {team.memberCount} {team.memberCount === 1 ? "member" : "members"}
   357→          </span>
   358→          <span className="inline-flex items-center gap-1.5">
   359→            <CheckSquare className="h-3.5 w-3.5" />
   360→            {team.taskCount} {team.taskCount === 1 ? "task" : "tasks"}
   361→          </span>
   362→          <span className="inline-flex items-center gap-1.5">
   363→            <CalendarDays className="h-3.5 w-3.5" />
   364→            {team.upcomingMeetingCount} upcoming
   365→          </span>
   366→        </div>
   367→      </div>
   368→    </div>
   369→  )
   370→}

   372→function TeamCardMenu({ clubId, team }: { clubId: string; team: Team }) {
   373→  const [editOpen, setEditOpen] = useState(false)
   374→  const [deleteOpen, setDeleteOpen] = useState(false)
   375→
   376→  const deleteMut = useMutation({
   377→    mutationFn: () => api(`/api/clubs/${clubId}/teams/${team.id}`, { method: "DELETE" }),
   378→    onSuccess: () => {
   379→      toast.success(`Team "${team.name}" deleted`)
   380→      setDeleteOpen(false)
   381→    },
   382→    onError: (e: Error) => toast.error(e.message),
   383→  })
   384→
   385→  return (
   386→    <>
   387→      <div onClick={(e) => e.stopPropagation()} role="presentation">
   388→        <DropdownMenu>
   389→          <DropdownMenuTrigger asChild>
   390→            <Button
   391→              variant="ghost"
   392→              size="icon"
   393→              className="h-7 w-7"
   394→              aria-label={`Actions for team ${team.name}`}
   395→            >
   396→              <MoreVertical className="h-4 w-4" />
   397→            </Button>
   398→          </DropdownMenuTrigger>
   399→          <DropdownMenuContent align="end">
   400→            <DropdownMenuItem onClick={() => setEditOpen(true)}>
   401→              <Pencil className="mr-2 h-4 w-4" /> Edit
   402→            </DropdownMenuItem>
   403→            <DropdownMenuSeparator />
   404→            <DropdownMenuItem
   405→              className="text-red-600 focus:text-red-700"
   406→              onClick={() => setDeleteOpen(true)}
   407→            >
   408→              <Trash2 className="mr-2 h-4 w-4" /> Delete
   409→            </DropdownMenuItem>
   410→          </DropdownMenuContent>
   411→        </DropdownMenu>
   412→      </div>
   413→      <EditTeamDialog
   414→        key={team.id}
   415→        clubId={clubId}
   416→        team={team}
   417→        open={editOpen}
   418→        onOpenChange={setEditOpen}
   419→      />
   420→      <DeleteTeamDialog
   421→        team={team}
   422→        open={deleteOpen}
   423→        onOpenChange={setDeleteOpen}
   424→        onConfirm={() => deleteMut.mutate()}
   425→        loading={deleteMut.isPending}
   426→      />
   427→    </>
   428→  )
   429→}

   435→function TeamDetailSheet({
   436→  clubId,
   437→  team,
   438→  isExec,
   439→  online,
   440→  onClose,
   441→}: {
   442→  clubId: string
   443→  team: Team
   444→  isExec: boolean
   445→  online: Set<string>
   446→  onClose: () => void
   447→}) {
   448→  const qc = useQueryClient()
   449→  const [addOpen, setAddOpen] = useState(false)
   450→  const [removeTarget, setRemoveTarget] = useState<TeamMember | null>(null)
   451→
   452→  const removeMut = useMutation({
   453→    mutationFn: (userId: string) =>
   454→      api(`/api/clubs/${clubId}/teams/${team.id}/members/${userId}`, { method: "DELETE" }),
   455→    onSuccess: () => {
   456→      toast.success("Member removed")
   457→      qc.invalidateQueries({ queryKey: ["teams", clubId] })
   458→      qc.invalidateQueries({ queryKey: ["members", clubId] })
   459→      setRemoveTarget(null)
   460→    },
   461→    onError: (e: Error) => toast.error(e.message),
   462→  })
   463→
   464→  return (
   465→    <>
   466→      <SheetHeader className="px-5 pt-5 pb-3 border-b">
   467→        <div className="flex items-start justify-between gap-2">
   468→          <div className="min-w-0">
   469→            <SheetTitle className="text-xl truncate">{team.name}</SheetTitle>
   470→            {team.description ? (
   471→              <SheetDescription className="mt-1 text-sm">{team.description}</SheetDescription>
   472→            ) : (
   473→              <SheetDescription className="mt-1 italic">No description</SheetDescription>
   474→            )}
   475→          </div>
   476→          <Button variant="ghost" size="icon" className="h-7 w-7 shrink-0" onClick={onClose} aria-label="Close">
   477→            <X className="h-4 w-4" />
   478→          </Button>
   479→        </div>
   480→        <div className="flex items-center gap-3 text-xs text-muted-foreground mt-2">
   481→          <span className="inline-flex items-center gap-1">
   482→            <Clock className="h-3.5 w-3.5" /> Created {relativeTime(team.createdAt)}
   483→          </span>
   484→        </div>
   485→      </SheetHeader>
   486→
   487→      <ScrollArea className="flex-1 min-h-0">
   488→        <div className="p-5 space-y-6">
   489→          {/* Roster */}
   490→          <section className="space-y-3">
   491→            <div className="flex items-center justify-between">
   492→              <h3 className="text-sm font-semibold flex items-center gap-2">
   493→                <Users className="h-4 w-4" /> Roster
   494→                <span className="text-muted-foreground font-normal">({team.memberCount})</span>
   495→              </h3>
   496→              {isExec && (
   497→                <Button size="sm" variant="outline" onClick={() => setAddOpen(true)}>
   498→                  <UserPlus className="mr-1.5 h-3.5 w-3.5" /> Add members
   499→                </Button>
   500→              )}
   501→            </div>
   502→            {team.members.length === 0 ? (
   503→              <div className="rounded-lg border border-dashed p-4 text-center text-sm text-muted-foreground">
   504→                {isExec ? "Add your first team member." : "No members yet."}
   505→              </div>
   506→            ) : (
   507→              <ul className="space-y-1.5">
   508→                {team.members.map((m) => (
   509→                  <li
   510→                    key={m.id}
   511→                    className="flex items-center gap-3 rounded-lg p-2 hover:bg-muted/50 transition-colors"
   512→                  >
   513→                    <Avatar className="h-8 w-8">
   514→                      <AvatarImage src={m.user.avatarUrl ?? undefined} alt={m.user.name} />
   515→                      <AvatarFallback className="text-xs">{initials(m.user.name)}</AvatarFallback>
   516→                      {online.has(m.userId) && (
   517→                        <span
   518→                          aria-label="Online"
   519→                          className="absolute bottom-0 right-0 h-2.5 w-2.5 rounded-full bg-club ring-2 ring-background"
   520→                        />
   521→                      )}
   522→                    </Avatar>
   523→                    <div className="min-w-0 flex-1">
   524→                      <div className="text-sm font-medium truncate">{m.user.name}</div>
   525→                      <div className="text-xs text-muted-foreground">
   526→                        Joined {relativeTime(m.joinedAt)}
   527→                      </div>
   528→                    </div>
   529→                    {isExec && (
   530→                      <Button
   531→                        variant="ghost"
   532→                        size="icon"
   533→                        className="h-7 w-7 text-muted-foreground hover:text-red-600"
   534→                        aria-label={`Remove ${m.user.name} from team`}
   535→                        onClick={() => setRemoveTarget(m)}
   536→                      >
   537→                        <UserMinus className="h-4 w-4" />
   538→                      </Button>
   539→                    )}
   540→                  </li>
   541→                ))}
   542→              </ul>
   543→            )}
   544→          </section>
   545→
   546→          <Separator />
   547→
   548→          {/* Tasks */}
   549→          <section className="space-y-3">
   550→            <h3 className="text-sm font-semibold flex items-center gap-2">

   556→                <Inbox className="h-5 w-5" />
   557→                No tasks assigned to this team.
   558→              </div>
   559→            ) : (
   560→              <ul className="space-y-1.5">
   561→                {team.tasks.map((t) => (
   562→                  <li
   563→                    key={t.id}
   564→                    className="flex items-center gap-3 rounded-lg p-2 hover:bg-muted/50 transition-colors"
   565→                  >
   566→                    <div className="min-w-0 flex-1">
   567→                      <div className="text-sm font-medium truncate">{t.title}</div>
   568→                      <div className="text-xs text-muted-foreground flex items-center gap-2">
   569→                        {t.assignee ? (
   570→                          <span className="inline-flex items-center gap-1">
   571→                            <Avatar className="h-4 w-4">
   572→                              <AvatarImage src={t.assignee.avatarUrl ?? undefined} alt={t.assignee.name} />
   573→                              <AvatarFallback className="text-[8px]">{initials(t.assignee.name)}</AvatarFallback>
   574→                            </Avatar>
   575→                            {t.assignee.name}
   576→                          </span>
   577→                        ) : (
   578→                          <span className="italic">Unassigned</span>
   579→                        )}
   580→                        {t.dueDate && (
   581→                          <span className="inline-flex items-center gap-1">
   582→                            · due {new Date(t.dueDate).toLocaleDateString()}
   583→                          </span>
   584→                        )}
   585→                      </div>
   586→                    </div>
   587→                    <TaskStatusBadge status={t.status} />
   588→                  </li>
   589→                ))}
   590→              </ul>
   591→            )}
   592→          </section>
   593→
   594→          <Separator />
   595→
   596→          {/* Upcoming meetings */}
   597→          <section className="space-y-3">
   598→            <h3 className="text-sm font-semibold flex items-center gap-2">
   599→              <CalendarDays className="h-4 w-4" /> Upcoming meetings
   600→              <span className="text-muted-foreground font-normal">({team.upcomingMeetingCount})</span>
   601→            </h3>

   643→      <AlertDialog open={!!removeTarget} onOpenChange={(o) => !o && setRemoveTarget(null)}>

   645→          <AlertDialogHeader>
   646→            <AlertDialogTitle>Remove member?</AlertDialogTitle>
   647→            <AlertDialogDescription>
   648→              Remove{" "}

   656→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   657→              onClick={() => removeTarget && removeMut.mutate(removeTarget.userId)}
   658→              disabled={removeMut.isPending}
   659→            >
   660→              {removeMut.isPending ? "Removing…" : "Remove"}
   661→            </AlertDialogAction>
   662→          </AlertDialogFooter>
   663→        </AlertDialogContent>
   664→      </AlertDialog>
   665→    </>
   666→  )

   669→function TaskStatusBadge({ status }: { status: string }) {

   672→    in_progress: "in_progress",
   673→    done: "done",
   674→  }
   675→  const kind = kindMap[status] ?? "neutral"
   676→  return <StatusBadge status={kind} />

   887→function AddMembersDialog({
   888→  clubId,
   889→  team,
   890→  open,
   891→  onOpenChange,
   892→}: {
   893→  clubId: string
   894→  team: Team
   895→  open: boolean
   896→  onOpenChange: (v: boolean) => void
   897→}) {
   898→  const qc = useQueryClient()
   899→  const [selected, setSelected] = useState<Set<string>>(new Set())
   900→  const [search, setSearch] = useState("")
   901→  const [loading, setLoading] = useState(false)
   902→
   903→  const { data: membersData, isLoading: membersLoading } = useQuery<MembersResponse>({
   904→    queryKey: ["members", clubId],
   905→    queryFn: () => api(`/api/clubs/${clubId}/members`),
   906→    enabled: !!clubId && open,
   907→  })
   908→
   909→  const existingIds = useMemo(
   910→    () => new Set(team.members.map((m) => m.userId)),
   911→    [team.members]
   912→  )
   913→
   914→  const candidates = useMemo(() => {
   915→    const list = membersData?.members ?? []
   916→    const q = search.trim().toLowerCase()
   917→    return list.filter((m) => {
   918→      if (existingIds.has(m.user.id)) return false
   919→      if (!q) return true
   920→      return (
   921→        m.user.name.toLowerCase().includes(q) ||
   922→        m.user.email.toLowerCase().includes(q)
   923→      )
   924→    })
   925→  }, [membersData, existingIds, search])
   926→
   927→  function reset() {
   928→    setSelected(new Set())
   929→    setSearch("")
   930→  }
   931→
   932→  async function submit() {
   933→    if (selected.size === 0) {
   934→      toast.error("Select at least one member")
   935→      return
   936→    }
   937→    setLoading(true)
   938→    let ok = 0
   939→    let fail = 0
   940→    for (const userId of Array.from(selected)) {
   941→      try {
   942→        await api(`/api/clubs/${clubId}/teams/${team.id}/members`, {
   943→          method: "POST",
   944→          json: { userId },
   945→        })
   946→        ok++
   947→      } catch {
   948→        fail++
   949→      }
   950→    }
   951→    setLoading(false)
   952→    if (ok > 0) {
   953→      toast.success(
   954→        `Added ${ok} ${ok === 1 ? "member" : "members"}${fail > 0 ? ` (${fail} failed)` : ""}`
   955→      )
   956→      qc.invalidateQueries({ queryKey: ["teams", clubId] })
   957→      qc.invalidateQueries({ queryKey: ["members", clubId] })
   958→    } else if (fail > 0) {
   959→      toast.error("Failed to add members")
   960→    }
   961→    reset()
   962→    onOpenChange(false)
   963→  }
   964→
   965→  function toggle(id: string) {
   966→    setSelected((prev) => {
   967→      const next = new Set(prev)
   968→      if (next.has(id)) next.delete(id)
   969→      else next.add(id)
   970→      return next
   971→    })
   972→  }
   973→
   974→  return (
   975→    <Dialog
   976→      open={open}
   977→      onOpenChange={(o) => {
   978→        if (!o) reset()
   979→        onOpenChange(o)
   980→      }}
   981→    >
   982→      <DialogContent className="max-w-md">
   983→        <DialogHeader>
   984→          <DialogTitle>Add members to {team.name}</DialogTitle>
   985→          <DialogDescription>
   986→            Select club members to add to this team. {team.memberCount} current{" "}
   987→            {team.memberCount === 1 ? "member" : "members"}.
   988→          </DialogDescription>
   989→        </DialogHeader>
   990→
   991→        <Input
   992→          placeholder="Search by name or email…"
   993→          value={search}
   994→          onChange={(e) => setSearch(e.target.value)}
   995→          aria-label="Search members"
   996→        />
   997→
   998→        <div className="max-h-72 overflow-y-auto rounded-lg border -mx-1">
   999→          {membersLoading ? (
  1000→            <div className="p-6 text-center text-sm text-muted-foreground">Loading members…</div>
  1001→          ) : candidates.length === 0 ? (
  1002→            <div className="p-6 text-center text-sm text-muted-foreground">
  1003→              {membersData && membersData.members.length === 0
  1004→                ? "No club members found."
  1005→                : "No members match your search or all are already on this team."}
  1006→            </div>
  1007→          ) : (
  1008→            <ul className="divide-y">
  1009→              {candidates.map((m) => {
  1010→                const checked = selected.has(m.user.id)
  1011→                return (
  1012→                  <li key={m.membershipId}>
  1013→                    <label
  1014→                      className={cn(
  1015→                        "flex items-center gap-3 px-3 py-2 cursor-pointer hover:bg-muted/50 transition-colors",
  1016→                        checked && "bg-club-muted"
  1017→                      )}
  1018→                    >
  1019→                      <Checkbox
  1020→                        checked={checked}
  1021→                        onCheckedChange={() => toggle(m.user.id)}
  1022→                        aria-label={`Select ${m.user.name}`}
  1023→                      />
  1024→                      <Avatar className="h-7 w-7">
  1025→                        <AvatarImage src={m.user.avatarUrl ?? undefined} alt={m.user.name} />
  1026→                        <AvatarFallback className="text-xs">{initials(m.user.name)}</AvatarFallback>
  1027→                      </Avatar>
  1028→                      <div className="min-w-0 flex-1">
  1029→                        <div className="text-sm font-medium truncate">{m.user.name}</div>
  1030→                        <div className="text-xs text-muted-foreground truncate">{m.user.email}</div>
  1031→                      </div>
  1032→                      {m.role === "executive" && (
  1033→                        <Badge variant="outline" className="text-[10px] px-1.5 py-0">
  1034→                          exec
  1035→                        </Badge>
  1036→                      )}
  1037→                    </label>
  1038→                  </li>
  1039→                )
  1040→              })}
  1041→            </ul>
  1042→          )}
  1043→        </div>
  1044→
  1045→        <DialogFooter className="flex items-center justify-between sm:justify-between gap-2">
  1046→          <span className="text-xs text-muted-foreground">{selected.size} selected</span>
  1047→          <div className="flex items-center gap-2">
  1048→            <Button variant="outline" onClick={() => onOpenChange(false)}>
  1049→              Cancel
  1050→            </Button>
  1051→            <Button
  1052→              variant="club"
  1053→              onClick={submit}
  1054→              disabled={loading || selected.size === 0}
  1055→            >
  1056→              {loading ? "Adding…" : `Add${selected.size > 0 ? ` ${selected.size}` : ""}`}
  1057→            </Button>
  1058→          </div>
  1059→        </DialogFooter>
  1060→      </DialogContent>
  1061→    </Dialog>
  1062→  )
  1063→}
