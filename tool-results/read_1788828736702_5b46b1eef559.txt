    16→} from "@/components/shared/page-header"

    28→// =========================================================================
    29→// Types — unchanged (matches the dashboard API response shape)
    30→// =========================================================================
    31→interface DashboardData {
    32→  club: {
    33→    id: string
    34→    name: string
    35→    description: string | null
    36→    accentColor: string
    37→    logoUrl: string | null
    38→    hoursGoal: number
    39→    memberCount: number
    40→    createdAt: string
    41→  }
    42→  myRole: "member" | "executive"
    43→  myStats: {
    44→    approvedHours: number
    45→    pendingHours: number
    46→    tasksAssigned: number
    47→    tasksDone: number
    48→    upcomingMeetings: number
    49→    myRsvpsGoing: number
    50→  }
    51→  clubStats: {
    52→    totalMembers: number
    53→    totalApprovedHours: number
    54→    pendingApprovals: number
    55→    openTasks: number
    56→    tasksDone: number
    57→    upcomingMeetingsCount: number
    58→    announcementsThisMonth: number
    59→    teamsCount: number
    60→  }
    61→  leaderboard: {
    62→    userId: string
    63→    name: string
    64→    avatarUrl: string | null
    65→    hours: number
    66→  }[]
    67→  recentAnnouncements: {
    68→    id: string
    69→    title: string
    70→    isPinned: boolean
    71→    createdAt: string
    72→    authorName: string
    73→  }[]
    74→  myTasks: {
    75→    id: string
    76→    title: string
    77→    dueDate: string | null
    78→    status: string
    79→    teamName: string | null
    80→  }[]
    81→  upcomingMeetings: {
    82→    id: string
    83→    title: string
    84→    startTime: string
    85→    location: string
    86→    myRsvp: "going" | "not_going" | "maybe" | null
    87→  }[]
    88→  hoursTrend: { date: string; hours: number }[]
    89→  execStats: {
    90→    avgApprovalTurnaroundHours: number | null
    91→    submissionsThisWeek: number
    92→    taskCompletionRate: number
    93→  } | null
    94→}
    95→
    96→/** Inline style for staggered fade-in cascade. */
    97→function stagger(i: number): CSSProperties {
    98→  return {
    99→    animationDelay: `${i * 50}ms`,
   100→    animationFillMode: "backwards",
   101→  }
   102→}

   104→// =========================================================================
   105→// Main component
   106→// =========================================================================
   107→export function DashboardView() {
   108→  const clubId = useAppStore((s) => s.currentClubId)
   109→  const setView = useAppStore((s) => s.setView)
   110→  const currentClub = useAppStore((s) => s.currentClub)
   111→  const [onboardingDismissed, setOnboardingDismissed] = useState(false)
   112→
   113→  // Read dismissed state from localStorage on club change.
   114→  // Deferred to a microtask so we don't call setState synchronously
   115→  // inside the effect body (avoids cascading renders).
   116→  useEffect(() => {
   117→    if (!clubId) return
   118→    let cancelled = false
   119→    Promise.resolve().then(() => {
   120→      if (cancelled) return
   121→      try {
   122→        const v = localStorage.getItem(`onboarding-dismissed-${clubId}`)
   123→        setOnboardingDismissed(v === "true")
   124→      } catch {
   125→        setOnboardingDismissed(false)
   126→      }
   127→    })
   128→    return () => {
   129→      cancelled = true
   130→    }
   131→  }, [clubId])
   132→
   133→  const { data, isLoading, error } = useQuery<DashboardData>({
   134→    queryKey: ["dashboard", clubId],
   135→    queryFn: () => api<DashboardData>(`/api/clubs/${clubId}/dashboard`),
   136→    enabled: !!clubId,
   137→    staleTime: 30_000,
   138→    // Realtime is primary; poll only as a fallback while the socket is down.
   139→    refetchInterval: usePollingFallback(15000),
   140→  })
   141→
   142→  useEffect(() => {
   143→    if (error) toast.error((error as Error).message || "Failed to load dashboard")
   144→  }, [error])
   145→
   146→  if (!clubId) {
   147→    return (
   148→      <EmptyState
   149→        icon={<Users className="h-8 w-8" />}
   150→        title="No club selected"
   151→        description="Pick a club from the sidebar to view its dashboard."
   152→      />
   153→    )
   154→  }
   155→
   156→  if (isLoading || !data) {
   157→    return <DashboardSkeleton />
   158→  }
   159→
   160→  const isExec = data.myRole === "executive"
   161→  const hasActivity =
   162→    data.recentAnnouncements.length > 0 || data.clubStats.teamsCount > 0
   163→  const showOnboarding = !onboardingDismissed && !hasActivity
   164→
   165→  const dismissOnboarding = () => {
   166→    try {
   167→      localStorage.setItem(`onboarding-dismissed-${clubId}`, "true")
   168→    } catch {
   169→      /* ignore */
   170→    }
   171→    setOnboardingDismissed(true)
   172→  }
   173→
   174→  // -- Tier 1 attention items ------------------------------------------------
   175→  const attentionItems: ReactNode[] = []
   176→  let idx = 0
   177→
   178→  if (isExec && data.clubStats.pendingApprovals > 0) {
   179→    const n = data.clubStats.pendingApprovals
   180→    attentionItems.push(
   181→      <AttentionCard
   182→        key="approvals"
   183→        delay={idx++ * 50}
   184→        icon={<ShieldCheck className="h-4 w-4" />}
   185→        title={`${n} pending approval${n === 1 ? "" : "s"}`}
   186→        description="Awaiting your review"
   187→        onClick={() => setView("approvals")}
   188→      />
   189→    )
   190→  }
   191→
   192→  if (data.myStats.pendingHours > 0) {
   193→    attentionItems.push(
   194→      <AttentionCard
   195→        key="hours"
   196→        delay={idx++ * 50}
   197→        icon={<Hourglass className="h-4 w-4" />}
   198→        title={`${fmtHours(data.myStats.pendingHours)}h awaiting review`}
   199→        description="Your submitted service hours"
   200→        onClick={() => setView("hours")}
   201→      />
   202→    )
   203→  }
   204→
   205→  // Tasks due soon / overdue
   206→  const tasksWithDue = data.myTasks
   207→    .filter((t) => t.status !== "done")
   208→    .map((t) => ({ ...t, due: t.dueDate ? new Date(t.dueDate) : null }))
   209→  const overdueTasks = tasksWithDue.filter(
   210→    (t) => t.due && isPast(t.due) && !isToday(t.due)
   211→  )
   212→  const dueSoon = tasksWithDue
   213→    .filter((t) => !t.due || !isPast(t.due) || isToday(t.due))
   214→    .sort((a, b) => {
   215→      if (!a.due) return 1
   216→      if (!b.due) return -1
   217→      return a.due.getTime() - b.due.getTime()
   218→    })
   219→    .slice(0, 3)
   220→  const topTasks = [...overdueTasks.slice(0, 2), ...dueSoon].slice(0, 3)
   221→
   222→  if (topTasks.length > 0) {
   223→    const n = overdueTasks.length
   224→    attentionItems.push(
   225→      <AttentionCard
   226→        key="tasks"
   227→        delay={idx++ * 50}
   228→        icon={<ClipboardList className="h-4 w-4" />}
   229→        title={n > 0 ? `${n} overdue task${n === 1 ? "" : "s"}` : "Tasks due soon"}
   230→        description={
   231→          n > 0
   232→            ? "Needs your attention"
   233→            : `${data.myStats.tasksAssigned} open task${data.myStats.tasksAssigned === 1 ? "" : "s"} assigned`
   234→        }
   235→        onClick={() => setView("tasks")}
   236→        urgent={n > 0}
   237→      >
   238→        <ul className="mt-2.5 space-y-1.5">
   239→          {topTasks.map((t) => {
   240→            const isOverdue = t.due && isPast(t.due) && !isToday(t.due)
   241→            return (
   242→              <li key={t.id} className="flex items-center gap-2 text-body">
   243→                {isOverdue ? (
   244→                  <span
   245→                    className="h-1.5 w-1.5 rounded-full bg-red-500 shrink-0"
   246→                    aria-label="Overdue"
   247→                  />
   248→                ) : (
   249→                  <span
   250→                    className="h-1.5 w-1.5 rounded-full bg-club shrink-0"
   251→                    aria-label="Upcoming"
   252→                  />
   253→                )}
   254→                <span className="truncate">{t.title}</span>
   255→                {t.due && (
   256→                  <span
   257→                    className={cn(
   258→                      "ml-auto text-caption shrink-0 tabular-nums",
   259→                      isOverdue && "text-red-600 dark:text-red-400 font-medium"
   260→                    )}
   261→                  >
   262→                    {format(t.due, "MMM d")}
   263→                  </span>
   264→                )}
   265→              </li>
   266→            )
   267→          })}
   268→        </ul>
   269→      </AttentionCard>
   270→    )
   271→  }
   272→
   273→  // Pinned announcements (last 1-2)
   274→  const pinned = data.recentAnnouncements.filter((a) => a.isPinned).slice(0, 2)
   275→  if (pinned.length > 0) {
   276→    attentionItems.push(
   277→      <AttentionCard
   278→        key="pinned"
   279→        delay={idx++ * 50}
   280→        icon={<Pin className="h-4 w-4" />}
   281→        title="Pinned announcement"
   282→        description={pinned.length > 1 ? `${pinned.length} pinned posts` : "From your club"}
   283→        onClick={() => setView("announcements")}
   284→      >
   285→        <ul className="mt-2.5 space-y-1.5">
   286→          {pinned.map((a) => (
   287→            <li key={a.id} className="text-body truncate">
   288→              {a.title}
   289→            </li>
   290→          ))}
   291→        </ul>
   292→      </AttentionCard>
   293→    )
   294→  }
   295→
   296→  // Hours goal progress (for snapshot)
   297→  const hoursPct =
   298→    data.club.hoursGoal > 0
   299→      ? Math.min(100, (data.myStats.approvedHours / data.club.hoursGoal) * 100)
   300→      : 0
   301→
   302→  return (
   303→    <div className="space-y-4 sm:space-y-5">
   304→      {/* Row 1 — Slim hero bar --------------------------------------------- */}
   305→      <HeroBar
   306→        name={data.club.name}
   307→        logoUrl={data.club.logoUrl}
   308→        memberCount={data.club.memberCount}
   309→        createdAt={data.club.createdAt}
   310→        role={data.myRole}
   311→      />
   312→
   313→      {/* Row 2 — Tier 1 (or onboarding banner) ----------------------------- */}
   314→      {showOnboarding ? (
   315→        <OnboardingBanner
   316→          clubCode={currentClub?.clubCode ?? null}
   317→          onDismiss={dismissOnboarding}
   318→          onNavigate={setView}
   319→          teamsCount={data.clubStats.teamsCount}
   320→          announcementsCount={
   321→            data.recentAnnouncements.length + data.clubStats.announcementsThisMonth
   322→          }
   323→          meetingsCount={data.clubStats.upcomingMeetingsCount}
   324→          memberCount={data.club.memberCount}
   325→        />
   326→      ) : attentionItems.length === 0 ? (
   327→        <AllCaughtUpCard />
   328→      ) : (
   329→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   330→          {attentionItems}
   331→        </div>
   332→      )}
   333→
   334→      {/* Rows 3+4 — chart + leaderboard + snapshot + announcements + meetings */}
   335→      <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
   336→        {/* Hours trend (Tier 3) — desktop: row 1, cols 1-8 */}
   337→        <section
   338→          className="card-quiet p-5 lg:col-span-8 order-2 lg:order-1 animate-fade-in"
   339→          style={stagger(2)}
   340→        >
   341→          <div className="flex items-center justify-between mb-4">
   342→            <div className="flex items-center gap-2">
   343→              <TrendingUp className="h-4 w-4 text-club" />
   344→              <h2 className="text-card-title">Approved hours · last 30 days</h2>
   345→            </div>
   346→            <Badge variant="secondary" className="text-xs tabular-nums">
   347→              Total {fmtHours(data.hoursTrend.reduce((s, d) => s + d.hours, 0))}h
   348→            </Badge>
   349→          </div>
   350→          {data.hoursTrend.every((d) => d.hours === 0) ? (
   351→            <EmptyState
   352→              icon={<BarChart3 className="h-8 w-8" />}
   353→              title="No hours logged yet"
   354→              description="Once hours are approved, you'll see a 30-day trend here."
   355→            />
   356→          ) : (
   357→            <div
   358→              className="h-56 w-full min-w-0"
   359→              aria-label="Approved hours trend chart"
   360→            >
   361→              <ResponsiveContainer width="100%" height="100%">
   362→                <AreaChart
   363→                  data={data.hoursTrend}
   364→                  margin={{ top: 8, right: 8, left: -16, bottom: 0 }}
   365→                >
   366→                  <defs>
   367→                    <linearGradient id="hoursGradient" x1="0" y1="0" x2="0" y2="1">
   368→                      <stop
   369→                        offset="0%"
   370→                        stopColor="var(--club-accent)"
   371→                        stopOpacity={0.4}
   372→                      />
   373→                      <stop
   374→                        offset="100%"
   375→                        stopColor="var(--club-accent)"
   376→                        stopOpacity={0.02}
   377→                      />
   378→                    </linearGradient>
   379→                  </defs>
   380→                  <CartesianGrid
   381→                    strokeDasharray="3 3"
   382→                    stroke="hsl(var(--border))"
   383→                    strokeOpacity={0.5}
   384→                    vertical={false}
   385→                  />
   386→                  <XAxis
   387→                    dataKey="date"
   388→                    tick={{ fontSize: 10, fill: "currentColor" }}
   389→                    tickFormatter={(d) => format(parseISO(d), "MMM d")}
   390→                    interval={Math.floor(data.hoursTrend.length / 6)}
   391→                    axisLine={false}
   392→                    tickLine={false}
   393→                    className="text-muted-foreground"
   394→                  />
   395→                  <YAxis
   396→                    tick={{ fontSize: 10, fill: "currentColor" }}
   397→                    allowDecimals={false}
   398→                    width={32}
   399→                    axisLine={false}
   400→                    tickLine={false}
   401→                    className="text-muted-foreground"
   402→                  />
   403→                  <Tooltip
   404→                    contentStyle={{
   405→                      borderRadius: 8,
   406→                      border: "1px solid hsl(var(--border, 220 13% 91%))",
   407→                      background: "hsl(var(--popover, 0 0% 100%))",
   408→                      color: "hsl(var(--popover-foreground, 0 0% 0%))",
   409→                      fontSize: 12,
   410→                    }}
   411→                    labelFormatter={(d) =>
   412→                      format(parseISO(String(d)), "MMM d, yyyy")
   413→                    }
   414→                    formatter={(value: number) => [
   415→                      `${fmtHours(value)}h`,
   416→                      "Approved hours",
   417→                    ]}
   418→                  />
   419→                  <Area
   420→                    type="monotone"
   421→                    dataKey="hours"
   422→                    stroke="var(--club-accent)"
   423→                    strokeWidth={2}
   424→                    fill="url(#hoursGradient)"
   425→                    dot={false}
   426→                  />
   427→                </AreaChart>
   428→              </ResponsiveContainer>
   429→            </div>
   430→          )}
   431→        </section>
   432→
   433→        {/* Leaderboard (Tier 3) — desktop: row 1, cols 9-12 */}
   434→        <section
   435→          className="card-quiet p-5 lg:col-span-4 order-3 lg:order-2 animate-fade-in"
   436→          style={stagger(3)}
   437→        >
   438→          <div className="flex items-center justify-between mb-3">
   439→            <div className="flex items-center gap-2">
   440→              <Trophy className="h-4 w-4 text-amber-500" />
   441→              <h2 className="text-card-title">Leaderboard</h2>
   442→            </div>
   443→            <Button
   444→              variant="ghost"
   445→              size="sm"
   446→              className="h-7 text-xs"
   447→              onClick={() => setView("members")}
   448→            >
   449→              All <ChevronRight className="h-3 w-3" />
   450→            </Button>
   451→          </div>
   452→          {data.leaderboard.length === 0 ? (
   453→            <EmptyState
   454→              icon={<Trophy className="h-8 w-8" />}
   455→              title="No hours logged yet"
   456→              description="Approved service hours will populate the leaderboard."
   457→            />
   458→          ) : (
   459→            <ul className="space-y-1">
   460→              {data.leaderboard.map((entry, i) => (
   461→                <li key={entry.userId}>
   462→                  <button
   463→                    type="button"
   464→                    onClick={() => setView("members")}
   465→                    className={cn(
   466→                      "flex items-center gap-3 w-full text-left rounded-lg px-2.5 py-2 hover:bg-accent/50 transition-colors focus:outline-none focus-visible:ring-2 focus-visible:ring-ring",
   467→                      i === 0 && "bg-club-muted/30 ring-1 ring-club/20"
   468→                    )}
   469→                  >
   470→                    <RankBadge rank={i + 1} />
   471→                    <Avatar className="h-8 w-8">
   472→                      <AvatarImage
   473→                        src={entry.avatarUrl ?? undefined}
   474→                        alt={entry.name}
   475→                      />
   476→                      <AvatarFallback
   477→                        className={cn("text-xs", avatarColor(entry.name))}
   478→                      >
   479→                        {initials(entry.name)}
   480→                      </AvatarFallback>
   481→                    </Avatar>
   482→                    <div className="min-w-0 flex-1">
   483→                      <p className="text-body-medium truncate">{entry.name}</p>
   484→                      <p className="text-caption tabular-nums">
   485→                        {fmtHours(entry.hours)} hours
   486→                      </p>
   487→                    </div>
   488→                  </button>
   489→                </li>
   490→              ))}
   491→            </ul>
   492→          )}
   493→        </section>
   494→
   495→        {/* Your snapshot (Tier 2) — desktop: row 2, cols 1-4; mobile: first */}
   496→        <section
   497→          className="card-quiet p-5 lg:col-span-4 order-1 lg:order-3 animate-fade-in"
   498→          style={stagger(0)}
   499→        >
   500→          <div className="flex items-center justify-between mb-4">
   501→            <div className="flex items-center gap-2">
   502→              <ListChecks className="h-4 w-4 text-club" />
   503→              <h2 className="text-card-title">Your snapshot</h2>
   504→            </div>
   505→            <Button
   506→              variant="ghost"
   507→              size="sm"
   508→              className="h-7 text-xs"
   509→              onClick={() => setView("hours")}
   510→            >
   511→              Hours <ChevronRight className="h-3 w-3" />
   512→            </Button>
   513→          </div>
   514→
   515→          {/* Approved hours + goal */}
   516→          <div className="space-y-2">
   517→            <div className="flex items-baseline justify-between">
   518→              <div className="flex items-center gap-2">
   519→                <Clock className="h-4 w-4 text-muted-foreground" />
   520→                <span className="text-body-medium">Approved hours</span>
   521→              </div>
   522→              <span className="text-card-title tabular-nums">
   523→                {fmtHours(data.myStats.approvedHours)}h
   524→              </span>
   525→            </div>
   526→            {data.club.hoursGoal > 0 && (
   527→              <>
   528→                <Progress
   529→                  value={hoursPct}
   530→                  className="h-1.5 [&_[data-slot=progress-indicator]]:bg-club"
   531→                />
   532→                <p className="text-caption tabular-nums">
   533→                  {fmtHours(data.myStats.approvedHours)} of {data.club.hoursGoal}h
   534→                  goal · {Math.round(hoursPct)}%
   535→                </p>
   536→              </>
   537→            )}
   538→          </div>
   539→
   540→          <Separator className="my-4" />
   541→
   542→          {/* Tasks */}
   543→          <div className="flex items-center justify-between">
   544→            <div className="flex items-center gap-2">
   545→              <ClipboardList className="h-4 w-4 text-muted-foreground" />
   546→              <span className="text-body-medium">Tasks</span>
   547→            </div>
   548→            <span className="text-caption-medium tabular-nums">
   549→              <span className="text-foreground font-medium">
   550→                {data.myStats.tasksAssigned}
   551→              </span>{" "}
   552→              open
   553→              <span className="mx-1 text-muted-foreground">/</span>
   554→              <span className="text-foreground font-medium">
   555→                {data.myStats.tasksDone}
   556→              </span>{" "}
   557→              done
   558→            </span>
   559→          </div>
   560→
   561→          <Separator className="my-4" />
   562→
   563→          {/* Next meetings */}
   564→          <div>
   565→            <div className="flex items-center gap-2 mb-2">
   566→              <CalendarDays className="h-4 w-4 text-muted-foreground" />
   567→              <span className="text-body-medium">Next meetings</span>
   568→            </div>
   569→            {data.upcomingMeetings.length === 0 ? (
   570→              <p className="text-caption">No upcoming meetings</p>
   571→            ) : (
   572→              <ul className="space-y-1.5">
   573→                {data.upcomingMeetings.slice(0, 3).map((m) => (
   574→                  <li
   575→                    key={m.id}
   576→                    className="flex items-center justify-between gap-2"
   577→                  >
   578→                    <span className="text-body truncate">{m.title}</span>
   579→                    <span className="text-caption shrink-0 tabular-nums">
   580→                      {format(new Date(m.startTime), "MMM d")}
   581→                    </span>
   582→                  </li>
   583→                ))}
   584→              </ul>
   585→            )}
   586→          </div>

  1098→function fmtHours(h: number) {
  1099→  if (h === 0) return "0"

  1104→function formatTurnaround(hours: number | null) {
  1105→  if (hours === null || hours === undefined) return "—"
  1106→  if (hours < 1) return `${Math.round(hours * 60)}m`
  1107→  if (hours < 24) return `${hours.toFixed(1)}h`
  1108→  return `${(hours / 24).toFixed(1)}d`
  1109→}
