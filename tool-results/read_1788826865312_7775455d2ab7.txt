   193→function linkify(text: string) {
   194→  // Combined URL + @mention renderer. Mentions get a styled span (no deep
   195→  // link target for v1 — just visual emphasis); URLs open in a new tab.
   196→  const parts: React.ReactNode[] = []
   197→  // Split on URL first to avoid running mention regex over links.
   198→  const urlSplit = text.split(URL_REGEX)
   199→  urlSplit.forEach((chunk, i) => {
   200→    if (/^https?:\/\//.test(chunk)) {
   201→      parts.push(
   202→        <a
   203→          key={`u${i}`}
   204→          href={chunk}
   205→          target="_blank"
   206→          rel="noopener noreferrer"
   207→          className="text-club hover:underline break-all"
   208→        >
   209→          {chunk}
   210→        </a>,
   211→      )
   212→    } else {
   213→      // Within this non-URL chunk, split on @mentions.
   214→      let lastIdx = 0
   215→      const re = new RegExp(MENTION_REGEX)
   216→      let m: RegExpExecArray | null
   217→      while ((m = re.exec(chunk)) !== null) {
   218→        if (m.index > lastIdx) {
   219→          parts.push(<span key={`t${i}-${lastIdx}`}>{chunk.slice(lastIdx, m.index)}</span>)
   220→        }
   221→        parts.push(
   222→          <span
   223→            key={`m${i}-${m.index}`}
   224→            className="text-club font-medium"
   225→          >
   226→            {m[0]}
   227→          </span>,
   228→        )
   229→        lastIdx = m.index + m[0].length
   230→      }
   231→      if (lastIdx < chunk.length) {
   232→        parts.push(<span key={`t${i}-end`}>{chunk.slice(lastIdx)}</span>)
   233→      }
   234→    }
   235→  })
   236→  return parts
   237→}

   253→export function ChatView() {
   254→  const clubId = useAppStore((s) => s.currentClubId)
   255→  // Delegate everything to an inner component keyed by clubId so React

   266→function ChatPane({ clubId }: { clubId: string }) {
   267→  const role = useAppStore((s) => s.currentClub?.role)
   268→  const isExec = role === "executive"
   269→  const { data: session } = useSession()
   270→  const myUserId = session?.user?.id
   271→
   272→  const [activeId, setActiveId] = useState<string | null>(null)
   273→  const [newGroupOpen, setNewGroupOpen] = useState(false)
   274→  const [newDmOpen, setNewDmOpen] = useState(false)
   275→  const [isMobileShowingMessages, setIsMobileShowingMessages] = useState(false)
   276→
   277→  // Hooks must be called unconditionally. online is empty when no club.
   278→  const online = usePresence(clubId)
   279→
   280→  // Load conversation list.
   281→  const conversationsQuery = useQuery<ConversationsResponse>({
   282→    queryKey: conversationsKey(clubId),
   283→    queryFn: () => api<ConversationsResponse>(`/api/clubs/${clubId}/chat/conversations`),
   284→    // Realtime is primary (chat_message events invalidate this key); poll
   285→    // only as a fallback while the socket is down.
   286→    refetchInterval: usePollingFallback(POLL_INTERVAL_MS),
   287→  })
   288→
   289→  const conversations = conversationsQuery.data?.conversations ?? []
   290→  const activeConversation = useMemo(
   291→    () => conversations.find((c) => c.id === activeId) ?? null,
   292→    [conversations, activeId],
   293→  )
   294→
   295→  // Auto-select the club-wide conversation if there is no active selection.
   296→  // Derived state — compute synchronously rather than via effect.
   297→  const effectiveActiveId =
   298→    activeId && conversations.some((c) => c.id === activeId)
   299→      ? activeId
   300→      : conversations.find((c) => c.type === "club_wide")?.id ?? null
   301→
   302→  // If we don't have an active id yet, pick the club-wide one (if any).
   303→  // This is a one-time set; once the user clicks a conversation, activeId
   304→  // is sticky until they switch clubs (which remounts this component via
   305→  // the key prop on the parent).
   306→  if (activeId === null && effectiveActiveId !== null) {
   307→    setActiveId(effectiveActiveId)
   308→  }
   309→
   310→  const openConversation = (id: string) => {
   311→    setActiveId(id)
   312→    setIsMobileShowingMessages(true)
   313→  }
   314→
   315→  const backToList = () => {
   316→    setIsMobileShowingMessages(false)
   317→  }
   318→
   319→  return (
   320→    <div className="h-[calc(100vh-7rem)] md:h-[calc(100vh-5rem)]">
   321→      <div className="flex h-full gap-0 md:gap-4">
   322→        {/* Sidebar — conversation list */}
   323→        <aside
   324→          className={cn(
   325→            "card-quiet w-full md:w-80 lg:w-96 flex-shrink-0 overflow-hidden flex flex-col",
   326→            isMobileShowingMessages && "hidden md:flex",
   327→          )}
   328→        >
   329→          <ConversationList
   330→            clubId={clubId}
   331→            conversations={conversations}
   332→            activeId={activeId}
   333→            onSelect={openConversation}
   334→            online={online}
   335→            isLoading={conversationsQuery.isLoading}
   336→            myUserId={myUserId ?? ""}
   337→            onNewGroup={() => setNewGroupOpen(true)}
   338→            onNewDm={() => setNewDmOpen(true)}
   339→          />
   340→        </aside>
   341→
   342→        {/* Main area — messages */}
   343→        <main
   344→          className={cn(
   345→            "card-quiet flex-1 overflow-hidden flex flex-col",
   346→            !isMobileShowingMessages && "hidden md:flex",
   347→          )}
   348→        >
   349→          {activeConversation ? (
   350→            <ConversationPane
   351→              key={activeConversation.id}
   352→              clubId={clubId}
   353→              conversation={activeConversation}
   354→              isExec={isExec}
   355→              myUserId={myUserId ?? ""}
   356→              online={online}
   357→              onBack={backToList}
   358→              onConversationMutated={() =>
   359→                conversationsQuery.refetch()
   360→              }
   361→            />
   362→          ) : (
   363→            <div className="flex-1 flex items-center justify-center p-6">
   364→              <EmptyState
   365→                icon={<MessageSquarePlus className="h-8 w-8" />}
   366→                title="No conversation selected"
   367→                description="Pick a conversation from the list, or start a new chat."
   368→              />
   369→            </div>
   370→          )}
   371→        </main>
   372→      </div>
   373→
   374→      {/* New group chat dialog */}
   375→      <NewGroupChatDialog
   376→        open={newGroupOpen}
   377→        onOpenChange={setNewGroupOpen}
   378→        clubId={clubId}
   379→        myUserId={myUserId ?? ""}
   380→        onCreated={(convId) => {
   381→          setNewGroupOpen(false)
   382→          void conversationsQuery.refetch().then(() => {
   383→            setActiveId(convId)
   384→            setIsMobileShowingMessages(true)
   385→          })
   386→        }}
   387→      />
   388→
   389→      {/* New DM dialog */}
   390→      <NewDirectChatDialog
   391→        open={newDmOpen}
   392→        onOpenChange={setNewDmOpen}
   393→        clubId={clubId}
   394→        myUserId={myUserId ?? ""}
   395→        onCreated={(convId) => {
   396→          setNewDmOpen(false)
   397→          void conversationsQuery.refetch().then(() => {
   398→            setActiveId(convId)
   399→            setIsMobileShowingMessages(true)
   400→          })
   401→        }}
   402→      />
   403→    </div>
   404→  )
   405→}

   411→function ConversationList({
   412→  clubId,
   413→  conversations,
   414→  activeId,
   415→  onSelect,
   416→  online,
   417→  isLoading,
   418→  myUserId,
   419→  onNewGroup,
   420→  onNewDm,
   421→}: {
   422→  clubId: string
   423→  conversations: ConversationItem[]
   424→  activeId: string | null
   425→  onSelect: (id: string) => void
   426→  online: Set<string>

   428→  myUserId: string
   429→  onNewGroup: () => void
   430→  onNewDm: () => void

   442→        c.type === "direct" ? c.otherUser?.name ?? "" : c.name ?? ""

   448→    <div className="flex flex-col h-full">
   449→      {/* Header */}
   450→      <div className="px-4 pt-4 pb-3 border-b border-border">

   453→          <div className="flex items-center gap-1">
   454→            <Tooltip>
   455→              <TooltipTrigger asChild>
   456→                <Button
   457→                  size="icon"
   458→                  variant="ghost"
   459→                  className="size-8"
   460→                  onClick={onNewDm}
   461→                  aria-label="New direct message"
   462→                >
   463→                  <MessageSquarePlus className="h-4 w-4" />
   464→                </Button>
   465→              </TooltipTrigger>
   466→              <TooltipContent>New direct message</TooltipContent>
   467→            </Tooltip>
   468→            <Tooltip>
   469→              <TooltipTrigger asChild>
   470→                <Button
   471→                  size="icon"
   472→                  variant="club"
   473→                  className="size-8"
   474→                  onClick={onNewGroup}
   475→                  aria-label="New group chat"
   476→                >
   477→                  <Plus className="h-4 w-4" />
   478→                </Button>
   479→              </TooltipTrigger>
   480→              <TooltipContent>New group chat</TooltipContent>
   481→            </Tooltip>
   482→          </div>
   483→        </div>
   484→        <div className="relative">
   485→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground pointer-events-none" />
   486→          <Input
   487→            value={search}
   488→            onChange={(e) => setSearch(e.target.value)}
   489→            placeholder="Search conversations…"
   490→            className="pl-8 h-8 text-sm"
   491→            aria-label="Search conversations"
   492→          />
   493→        </div>
   494→      </div>
   495→
   496→      {/* List */}
   497→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   498→        {isLoading ? (
   499→          <ConversationListSkeleton />
   500→        ) : conversations.length === 0 ? (
   501→          <div className="p-4">
   502→            <EmptyState
   503→              icon={<MessageSquarePlus className="h-8 w-8" />}
   504→              title="No conversations yet"
   505→              description="Start a direct message or create a group chat."
   506→              action={
   507→                <Button variant="club" size="sm" onClick={onNewDm}>
   508→                  <Plus className="h-4 w-4" /> Start a conversation
   509→                </Button>
   510→              }
   511→            />
   512→          </div>
   513→        ) : (
   514→          <>
   515→            {/* Club-wide section */}
   516→            {clubWide.length > 0 && (
   517→              <div>
   518→                <div className="px-3 pt-3 pb-1">
   519→                  <span className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   520→                    Club-wide
   521→                  </span>
   522→                </div>
   523→                {clubWide.map((c) => (
   524→                  <ConversationRow
   525→                    key={c.id}
   526→                    conversation={c}
   527→                    active={c.id === activeId}
   528→                    onClick={() => onSelect(c.id)}
   529→                    online={online}
   530→                    myUserId={myUserId}
   531→                  />
   532→                ))}
   533→                <div className="border-b border-border mx-3 my-1" />
   534→              </div>
   535→            )}
   536→
   537→            {/* Other conversations */}
   538→            <div className="pb-2">
   539→              {filteredOthers.length === 0 ? (
   540→                <div className="text-caption text-center py-6 px-4">
   541→                  {search ? "No matches." : "No conversations yet."}
   542→                </div>
   543→              ) : (
   544→                filteredOthers.map((c) => (
   545→                  <ConversationRow
   546→                    key={c.id}
   547→                    conversation={c}
   548→                    active={c.id === activeId}
   549→                    onClick={() => onSelect(c.id)}
   550→                    online={online}
   551→                    myUserId={myUserId}
   552→                  />
   553→                ))
   554→              )}
   555→            </div>
   556→          </>

   558→      </div>
   559→    </div>
   560→  )
   561→}

   563→function ConversationRow({
   564→  conversation,
   565→  active,
   566→  onClick,
   567→  online,
   568→  myUserId,
   569→}: {
   570→  conversation: ConversationItem
   571→  active: boolean
   572→  onClick: () => void
   573→  online: Set<string>
   574→  myUserId: string
   575→}) {
   576→  const title = conversationTitle(conversation)
   577→  const isDirect = conversation.type === "direct"
   578→  const otherId = conversation.otherUser?.id
   579→  const isOnline = !!otherId && online.has(otherId)
   580→  const initialsText = isDirect
   581→    ? conversation.otherUser?.name
   582→    : conversation.name
   583→
   584→  return (
   585→    <button
   586→      type="button"
   587→      onClick={onClick}
   588→      aria-current={active ? "true" : undefined}
   589→      className={cn(
   590→        "w-full text-left px-3 py-2.5 flex items-center gap-3 transition-colors hover:bg-accent/60 dark:hover:bg-accent/40 focus-visible:bg-accent/60",
   591→        active && "bg-club-muted hover:bg-club-muted",
   592→      )}
   593→    >
   594→      <div className="relative shrink-0">
   595→        {isDirect ? (
   596→          <Avatar className="size-9">
   597→            <AvatarImage src={conversation.otherUser?.avatarUrl ?? undefined} alt={title} />
   598→            <AvatarFallback className={cn("text-xs", avatarColor(initialsText))}>
   599→              {initials(initialsText)}
   600→            </AvatarFallback>
   601→          </Avatar>
   602→        ) : (
   603→          <div
   604→            className={cn(
   605→              "size-9 rounded-full flex items-center justify-center text-xs font-medium",
   606→              avatarColor(initialsText),
   607→            )}
   608→          >
   609→            {initials(initialsText)}
   610→          </div>
   611→        )}
   612→        {isOnline && (
   613→          <span
   614→            className="absolute bottom-0 right-0 size-2.5 rounded-full bg-club ring-2 ring-card"
   615→            aria-label="Online"
   616→          />
   617→        )}
   618→      </div>
   619→      <div className="flex-1 min-w-0">
   620→        <div className="flex items-baseline justify-between gap-2">
   621→          <span
   622→            className={cn(
   623→              "text-sm font-medium truncate",
   624→              active ? "text-foreground" : "text-foreground/90",
   625→            )}
   626→          >
   627→            {title}
   628→          </span>
   629→          {conversation.lastMessage && (
   630→            <span className="text-[10px] text-muted-foreground shrink-0">
   631→              {relativeTime(conversation.lastMessage.createdAt)}
   632→            </span>
   633→          )}
   634→        </div>
   635→        <div className="flex items-center justify-between gap-2 mt-0.5">
   636→          <span className="text-xs text-muted-foreground truncate">
   637→            {conversation.lastMessage ? (
   638→              <>
   639→                {conversation.lastMessage.authorId === myUserId ? (
   640→                  <span className="text-muted-foreground/80">You: </span>
   641→                ) : conversation.type !== "direct" && conversation.lastMessage.authorName ? (
   642→                  <span className="text-muted-foreground/80">
   643→                    {conversation.lastMessage.authorName.split(" ")[0]}:{" "}
   644→                  </span>
   645→                ) : null}
   646→                {truncate(conversation.lastMessage.body, 40)}
   647→              </>
   648→            ) : (
   649→              <span className="italic text-muted-foreground/70">No messages yet</span>
   650→            )}
   651→          </span>
   652→          {conversation.unreadCount > 0 && (
   653→            <span className="shrink-0 inline-flex items-center justify-center min-w-4 h-4 px-1 rounded-full bg-club text-club-foreground text-[10px] font-semibold tabular-nums">
   654→              {conversation.unreadCount > 99 ? "99+" : conversation.unreadCount}
   655→            </span>
   656→          )}
   657→        </div>
   658→      </div>
   659→    </button>
   660→  )
   661→}
   662→
   663→function truncate(text: string, max: number) {
   664→  if (text.length <= max) return text
   665→  return text.slice(0, max - 1) + "…"
   666→}

   672→function ConversationPane({
   673→  clubId,
   674→  conversation,
   675→  isExec,
   676→  myUserId,
   677→  online,
   678→  onBack,
   679→  onConversationMutated,
   680→}: {
   681→  clubId: string
   682→  conversation: ConversationItem
   683→  isExec: boolean
   684→  myUserId: string
   685→  online: Set<string>
   686→  onBack: () => void
   687→  onConversationMutated: () => void
   688→}) {
   689→  const qc = useQueryClient()
   690→  const [renameOpen, setRenameOpen] = useState(false)
   691→  const [addMemberOpen, setAddMemberOpen] = useState(false)
   692→  const [leaveOpen, setLeaveOpen] = useState(false)
   693→
   694→  // Latest page (latest PAGE_SIZE messages).
   695→  const latestQuery = useQuery<MessagesResponse>({
   696→    queryKey: messagesKey(clubId, conversation.id),
   697→    queryFn: () =>
   698→      api<MessagesResponse>(
   699→        `/api/clubs/${clubId}/chat/conversations/${conversation.id}/messages`,
   700→      ),
   701→    // Realtime is primary; poll only as a fallback while the socket is down.
   702→    refetchInterval: usePollingFallback(POLL_INTERVAL_MS),
   703→  })
   704→
   705→  // Older pages (loaded on scroll-up).
   706→  const [olderPages, setOlderPages] = useState<MessageItem[][]>([])
   707→  const [loadingOlder, setLoadingOlder] = useState(false)
   708→  const [hasMoreOlder, setHasMoreOlder] = useState(true)
   709→  const scrollContainerRef = useRef<HTMLDivElement | null>(null)
   710→  const prevScrollHeightRef = useRef<number>(0)
   711→  const pinnedBottomRef = useRef<boolean>(true)
   712→  // NOTE: This component is keyed by conversation.id in the parent, so React
   713→  // already remounts it (resetting local state) when switching conversations.
   714→
   715→  const latestMessages = latestQuery.data?.messages ?? []
   716→  const olderMessages = useMemo(() => olderPages.flat(), [olderPages])
   717→
   718→  // Dedupe: drop older messages that overlap with the latest window.
   719→  const latestIds = useMemo(() => new Set(latestMessages.map((m) => m.id)), [latestMessages])
   720→  const allMessages = useMemo(() => {
   721→    const older = olderMessages.filter((m) => !latestIds.has(m.id))
   722→    return [...older, ...latestMessages]
   723→  }, [olderMessages, latestIds, latestMessages])
   724→
   725→  // Auto-scroll to bottom on new messages (if user was already at the bottom).
   726→  useEffect(() => {
   727→    const el = scrollContainerRef.current
   728→    if (!el) return
   729→    if (pinnedBottomRef.current) {
   730→      el.scrollTop = el.scrollHeight
   731→    }
   732→  }, [allMessages.length, conversation.id])
   733→
   734→  // When older messages are prepended, preserve the visible scroll position.
   735→  useEffect(() => {
   736→    const el = scrollContainerRef.current
   737→    if (!el) return
   738→    if (prevScrollHeightRef.current && prevScrollHeightRef.current > 0) {
   739→      const delta = el.scrollHeight - prevScrollHeightRef.current
   740→      el.scrollTop = el.scrollTop + delta
   741→      prevScrollHeightRef.current = 0
   742→    }
   743→  }, [olderPages])
   744→
   745→  const handleScroll = useCallback(() => {
   746→    const el = scrollContainerRef.current
   747→    if (!el) return
   748→    const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight
   749→    pinnedBottomRef.current = distanceFromBottom < 80
   750→    if (el.scrollTop <= 40 && !loadingOlder && hasMoreOlder && latestMessages.length > 0) {
   751→      void loadOlder()
   752→    }
   753→  }, [loadingOlder, hasMoreOlder, latestMessages.length])
   754→
   755→  async function loadOlder() {
   756→    if (loadingOlder || !hasMoreOlder) return
   757→    const oldest = allMessages[0]
   758→    if (!oldest) return
   759→    setLoadingOlder(true)
   760→    const el = scrollContainerRef.current
   761→    if (el) prevScrollHeightRef.current = el.scrollHeight
   762→    try {
   763→      const res = await api<MessagesResponse>(
   764→        `/api/clubs/${clubId}/chat/conversations/${conversation.id}/messages?before=${oldest.id}`,
   765→      )
   766→      if (res.messages.length === 0) {
   767→        setHasMoreOlder(false)
   768→      } else {
   769→        setOlderPages((prev) => [res.messages, ...prev])
   770→        if (!res.hasMore) setHasMoreOlder(false)
   771→      }
   772→    } catch (e) {
   773→      // silent
   774→    } finally {
   775→      setLoadingOlder(false)
   776→    }
   777→  }
   778→
   779→  function patchMessage(messageId: string, patch: Partial<MessageItem>) {
   780→    qc.setQueryData<MessagesResponse>(messagesKey(clubId, conversation.id), (old) => {
   781→      if (!old) return old
   782→      return {
   783→        ...old,
   784→        messages: old.messages.map((m) =>
   785→          m.id === messageId ? { ...m, ...patch } : m,
   786→        ),
   787→      }
   788→    })

   790→      pages.map((page) =>
   791→        page.map((m) => (m.id === messageId ? { ...m, ...patch } : m)),
   792→      ),
   793→    )
   794→  }

   801→    patchMessage(messageId, { reactions, myReaction })

   803→
   804→  // Send message mutation.

   807→      api<{ message: MessageItem }>(
   808→        `/api/clubs/${clubId}/chat/conversations/${conversation.id}/messages`,
   809→        { method: "POST", json: { body } },
   810→      ),

   813→        if (!old) return { messages: [res.message], hasMore: false }
   814→        // Append if not present.
   815→        if (old.messages.some((m) => m.id === res.message.id)) return old
   816→        return { ...old, messages: [...old.messages, res.message] }
   817→      })
   818→      pinnedBottomRef.current = true

   847→    onSuccess: (_res, messageId) => {

   851→    onError: (e: Error) => toast.error(e.message),

   878→      void qc.invalidateQueries({ queryKey: pinnedKey(clubId, conversation.id) })

  1006→          <button

  1008→            onClick={() => void loadOlder()}

  1010→          >
  1011→            Load older messages

  1074→            onConversationMutated()
  1075→          }}
  1076→        />
  1077→      )}
  1078→
  1079→      {/* Add member dialog */}
  1080→      {addMemberOpen && (
  1081→        <AddMemberDialog
  1082→          open={addMemberOpen}
  1083→          onOpenChange={setAddMemberOpen}
  1084→          clubId={clubId}
  1085→          conversationId={conversation.id}
  1086→          onAdded={() => {
  1087→            setAddMemberOpen(false)
  1088→            void qc.invalidateQueries({ queryKey: conversationsKey(clubId) })
  1089→            onConversationMutated()
  1090→          }}
  1091→        />
  1092→      )}
  1093→
  1094→      {/* Leave confirm */}
  1095→      <AlertDialog open={leaveOpen} onOpenChange={setLeaveOpen}>
  1096→        <AlertDialogContent>
  1097→          <AlertDialogHeader>
  1098→            <AlertDialogTitle>Leave this conversation?</AlertDialogTitle>
  1099→            <AlertDialogDescription>
  1100→              You will no longer receive messages from this group chat. You can
  1101→              be re-added later by an owner or executive.
  1102→            </AlertDialogDescription>
  1103→          </AlertDialogHeader>
  1104→          <AlertDialogFooter>
  1105→            <AlertDialogCancel disabled={leaveMutation.isPending}>Cancel</AlertDialogCancel>
  1106→            <AlertDialogAction
  1107→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
  1108→              disabled={leaveMutation.isPending}
  1109→              onClick={(e) => {
  1110→                e.preventDefault()
  1111→                leaveMutation.mutate()
  1112→              }}
  1113→            >
  1114→              {leaveMutation.isPending ? (
  1115→                <Loader2 className="h-4 w-4 animate-spin" />
  1116→              ) : (
  1117→                "Leave"
  1118→              )}
  1119→            </AlertDialogAction>
  1120→          </AlertDialogFooter>
  1121→        </AlertDialogContent>
  1122→      </AlertDialog>
  1123→    </div>
  1124→  )
  1125→}

  1173→function MembersDropdown({

  1179→  conversationId: string

  1181→}) {

  1184→    queryFn: () => api<MemberListResponse>(`/api/clubs/${clubId}/members`),

  1190→  // show the full club member list filtered to those online. This is a
  1191→  // lightweight display aid — to be enhanced.

  1195→  return (

  1346→      currentGroup.push(m)

  1348→      if (currentGroup.length > 0) grouped.push(currentGroup)

  1667→/* =========================================================================

  1669→   ========================================================================= */
  1670→
  1671→/**
  1672→ * Shows "X is typing…" / "Several people are typing…" below the conversation
  1673→ * header. Subscribes directly to `realtime:club` events with type `chat_typing`
  1674→ * (and `chat_typing:stop`) — we deliberately DON'T invalidate React-Query here
  1675→ * because typing is ephemeral and shouldn't trigger message refetches.
  1676→ *
  1677→ * Each typing entry expires after TYPING_TTL_MS so a disconnected user
  1678→ * doesn't stay "typing" forever.
  1679→ */
  1680→function TypingIndicator({
  1681→  conversationId,
  1682→  myUserId,
  1683→}: {
  1684→  conversationId: string
  1685→  myUserId: string
  1686→}) {
  1687→  const clubId = useAppStore((s) => s.currentClubId)
  1688→  // Map<userId, expiresAt>
  1689→  const [typingUsers, setTypingUsers] = useState<Map<string, number>>(() => new Map())
  1690→
  1691→  useEffect(() => {
  1692→    const bump = (userId: string, expiresAt: number) => {
  1693→      setTypingUsers((prev) => {
  1694→        const next = new Map(prev)
  1695→        next.set(userId, expiresAt)
  1696→        return next
  1697→      })
  1698→    }
  1699→    const remove = (userId: string) => {
  1700→      setTypingUsers((prev) => {
  1701→        if (!prev.has(userId)) return prev
  1702→        const next = new Map(prev)
  1703→        next.delete(userId)
  1704→        return next
  1705→      })
  1706→    }
  1707→
  1708→    const onClub = (data: any) => {
  1709→      if (!data || data.conversationId !== conversationId) return
  1710→      if (data.userId === myUserId) return // ignore self
  1711→      if (data.type === "chat_typing") {
  1712→        bump(data.userId, Date.now() + TYPING_TTL_MS)
  1713→      } else if (data.type === "chat_typing:stop") {
  1714→        remove(data.userId)
  1715→      }
  1716→    }
  1717→    const unsub = onRealtimeEvent("realtime:club", onClub)
  1718→
  1719→    // Sweep expired typing entries every second so a stale "X is typing…"
  1720→    // doesn't linger if the user stops typing without sending a stop event
  1721→    // (e.g. they closed their tab).
  1722→    const sweeper = setInterval(() => {
  1723→      setTypingUsers((prev) => {
  1724→        const now = Date.now()
  1725→        let changed = false
  1726→        const next = new Map<string, number>()
  1727→        for (const [uid, expiresAt] of prev) {
  1728→          if (expiresAt > now) next.set(uid, expiresAt)
  1729→          else changed = true
  1730→        }
  1731→        return changed ? next : prev
  1732→      })
  1733→    }, 1000)
  1734→
  1735→    return () => {
  1736→      unsub()
  1737→      clearInterval(sweeper)
  1738→    }
  1739→  }, [conversationId, myUserId])
  1740→
  1741→  // Look up display names for the typing users. We use the global members
  1742→  // list (cached via React-Query elsewhere) so we don't fire another fetch
  1743→  // here. Stale data is fine — names rarely change.
  1744→  const { data: membersData } = useQuery<MemberListResponse>({
  1745→    queryKey: ["members", clubId ?? ""],
  1746→    queryFn: () => api<MemberListResponse>(`/api/clubs/${clubId ?? ""}/members`),
  1747→    enabled: !!clubId && typingUsers.size > 0,
  1748→    staleTime: 60_000,
  1749→  })
  1750→  const nameFor = (userId: string) =>
  1751→    membersData?.members.find((m) => m.user.id === userId)?.user.name ?? "Someone"
  1752→
  1753→  const others = Array.from(typingUsers.keys())
  1754→  if (others.length === 0) return null
  1755→  const label =
  1756→    others.length === 1
  1757→      ? `${nameFor(others[0])} is typing…`

  1784→function MessageComposer({
  1785→  clubId,
  1786→  conversationId,
  1787→  myUserId,
  1788→  onSend,
  1789→  disabled,
  1790→}: {
  1791→  clubId: string
  1792→  conversationId: string
  1793→  myUserId: string
  1794→  onSend: (body: string) => void
  1795→  disabled: boolean
  1796→}) {
  1797→  const [value, setValue] = useState("")
  1798→  const textareaRef = useRef<HTMLTextAreaElement | null>(null)
  1799→  const lastTypingAtRef = useRef<number>(0)
  1800→
  1801→  // Load the club members list (cached) to power the @mention picker.
  1802→  const { data: membersData } = useQuery<MemberListResponse>({
  1803→    queryKey: ["members", clubId],
  1804→    queryFn: () => api<MemberListResponse>(`/api/clubs/${clubId}/members`),
  1805→    staleTime: 30_000,
  1806→  })
  1807→  const members: MentionableMember[] = (membersData?.members ?? [])
  1808→    .filter((m) => m.user.id !== myUserId)
  1809→    .map((m) => ({
  1810→      userId: m.user.id,
  1811→      name: m.user.name,
  1812→      avatarUrl: m.user.avatarUrl ?? null,
  1813→    }))
  1814→
  1815→  // Auto-grow.
  1816→  useEffect(() => {
  1817→    const el = textareaRef.current
  1818→    if (!el) return
  1819→    el.style.height = "auto"
  1820→    el.style.height = Math.min(el.scrollHeight, 160) + "px"
  1821→  }, [value])
  1822→
  1823→  // Emit a typing event (debounced). We throttle to one event per
  1824→  // TYPING_DEBOUNCE_MS so the server isn't spammed on every keystroke.
  1825→  function maybeEmitTyping() {
  1826→    const now = Date.now()
  1827→    if (now - lastTypingAtRef.current < TYPING_DEBOUNCE_MS) return
  1828→    lastTypingAtRef.current = now
  1829→    try {
  1830→      const socket = getRealtimeSocket()
  1831→      if (socket.connected) {
  1832→        socket.emit("typing", { clubId, conversationId, userId: myUserId })
  1833→      }
  1834→    } catch {
  1835→      // ignore — typing indicators are best-effort
  1836→    }
  1837→  }
  1838→
  1839→  function submit() {
  1840→    const body = value.trim()
  1841→    if (!body || disabled) return
  1842→    onSend(body)
  1843→    setValue("")
  1844→    // Clear typing on send (so other clients stop showing "is typing" after
  1845→    // the message arrives). We send a stop event so the indicator clears
  1846→    // immediately rather than after the TTL expires.
  1847→    try {
  1848→      const socket = getRealtimeSocket()
  1849→      if (socket.connected) {
  1850→        socket.emit("typing:stop", { clubId, conversationId, userId: myUserId })
  1851→      }
  1852→    } catch {
  1853→      // ignore
  1854→    }
  1855→  }
  1856→
  1857→  return (
  1858→    <div className="border-t border-border p-3 md:p-4">
  1859→      <div className="flex items-end gap-2">
  1860→        <MentionableTextarea
  1861→          ref={textareaRef}
  1862→          value={value}
  1863→          members={members}
  1864→          onChange={(e) => {
  1865→            setValue(e.target.value)
  1866→            maybeEmitTyping()
  1867→          }}
  1868→          placeholder="Type a message… use @ to mention someone"
  1869→          className="flex-1 resize-none min-h-[40px] max-h-40"
  1870→          rows={1}
  1871→          onKeyDown={(e) => {
  1872→            if (e.key === "Enter" && !e.shiftKey) {
  1873→              e.preventDefault()
  1874→              submit()
  1875→            }
  1876→          }}
  1877→          aria-label="Message input"
  1878→          disabled={disabled}
  1879→          listLabel="Mention a club member"
  1880→        />
  1881→        <Button
  1882→          variant="club"
  1883→          size="icon"
  1884→          className="size-10 shrink-0"
  1885→          onClick={submit}
  1886→          disabled={disabled || !value.trim()}
  1887→          aria-label="Send message"
  1888→        >
  1889→          {disabled ? (
  1890→            <Loader2 className="h-4 w-4 animate-spin" />

  1892→            <Send className="h-4 w-4" />

  1894→        </Button>
  1895→      </div>
  1896→      <p className="text-[10px] text-muted-foreground mt-1.5 px-1">

  1898→        <kbd className="rounded border border-border px-1">Shift+Enter</kbd> for a new line.
  1899→        Type <kbd className="rounded border border-border px-1">@</kbd> to mention someone.
  1900→      </p>
  1901→    </div>
