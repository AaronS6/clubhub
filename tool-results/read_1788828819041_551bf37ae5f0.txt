   303→/**
   304→ * Dismissible banner shown at the very top of the app whenever the current
   305→ * club has an active urgent announcement. Dismissal is stored in localStorage
   306→ * per-announcement (`urgent-dismissed-<id>`) and auto-expires after 7 days.
   307→ *
   308→ * Only the LATEST urgent announcement is shown — older urgents effectively
   309→ * retire themselves by being superseded (the API returns only the most
   310→ * recent one), so a new urgent clears the dismissed state by virtue of
   311→ * having a different ID.
   312→ */
   313→const URGENT_DISMISS_TTL_MS = 7 * 24 * 60 * 60 * 1000 // 7 days
   314→
   315→function UrgentBanner() {
   316→  const currentClubId = useAppStore((s) => s.currentClubId)
   317→  const setView = useAppStore((s) => s.setView)
   318→  const [dismissedId, setDismissedId] = useState<string | null>(null)
   319→
   320→  const { data } = useQuery<{ announcement: {
   321→    id: string
   322→    title: string
   323→    body: string
   324→    createdAt: string
   325→    author: { id: string; name: string }
   326→  } | null }>({
   327→    queryKey: ["urgent-announcement", currentClubId ?? ""],
   328→    queryFn: () => api(`/api/clubs/${currentClubId}/announcements/urgent`),
   329→    enabled: !!currentClubId,
   330→    // Poll every 60s — urgent banners are high-signal; we want a near-real-time
   331→    // surfacing of new urgent announcements without spamming the server.
   332→    refetchInterval: 60_000,
   333→    staleTime: 30_000,
   334→  })
   335→
   336→  const announcement = data?.announcement ?? null
   337→
   338→  // Read dismissal state from localStorage when the announcement changes.
   339→  // Deferred to a microtask so we don't call setState synchronously inside
   340→  // the effect body (which would trigger cascading renders per React 19's
   341→  // react-hooks/set-state-in-effect rule).
   342→  useEffect(() => {
   343→    let cancelled = false
   344→    Promise.resolve().then(() => {
   345→      if (cancelled) return
   346→      if (!announcement) {
   347→        setDismissedId(null)
   348→        return
   349→      }
   350→      try {
   351→        const key = `urgent-dismissed-${announcement.id}`
   352→        const raw = localStorage.getItem(key)
   353→        if (!raw) {
   354→          setDismissedId(null)
   355→          return
   356→        }
   357→        const parsed = JSON.parse(raw) as { expiresAt?: number }
   358→        if (parsed.expiresAt && parsed.expiresAt > Date.now()) {
   359→          setDismissedId(announcement.id)
   360→        } else {
   361→          // expired — clear it so the banner re-shows
   362→          localStorage.removeItem(key)
   363→          setDismissedId(null)
   364→        }
   365→      } catch {
   366→        setDismissedId(null)
   367→      }
   368→    })
   369→    return () => {
   370→      cancelled = true
   371→    }
   372→  }, [announcement])
   373→
   374→  // Realtime refresh: when an announcement is created/updated/deleted, the
   375→  // urgent endpoint should be re-fetched so the banner appears/disappears
   376→  // immediately rather than waiting for the next poll.
   377→  const qc = useQueryClient()
   378→  useEffect(() => {
   379→    const unsub = onRealtimeEvent("realtime:club", (d: any) => {
   380→      const t = d?.type as string | undefined
   381→      if (
   382→        t === "announcement_created" ||
   383→        t === "announcement_updated" ||
   384→        t === "announcement_deleted" ||
   385→        t === "new_announcement"
   386→      ) {
   387→        qc.invalidateQueries({ queryKey: ["urgent-announcement", currentClubId ?? ""] })
   388→      }
   389→    })
   390→    return () => unsub()
   391→  }, [currentClubId, qc])
   392→
   393→  function dismiss() {
   394→    if (!announcement) return
   395→    const expiresAt = Date.now() + URGENT_DISMISS_TTL_MS
   396→    try {
   397→      localStorage.setItem(
   398→        `urgent-dismissed-${announcement.id}`,
   399→        JSON.stringify({ expiresAt }),
   400→      )
   401→    } catch {
   402→      // ignore — localStorage may be unavailable in private browsing
   403→    }
   404→    setDismissedId(announcement.id)
   405→  }
   406→
   407→  if (!announcement || dismissedId === announcement.id) return null
   408→
   409→  const bodyPreview = announcement.body.length > 120
   410→    ? announcement.body.slice(0, 117) + "…"
   411→    : announcement.body
   412→
   413→  return (
   414→    <div
   415→      role="alert"
   416→      className="flex items-center gap-3 px-3 sm:px-4 py-2 border-b border-red-200 dark:border-red-900/60 bg-red-50 dark:bg-red-950/30 text-red-900 dark:text-red-100 animate-fade-in"
   417→    >
   418→      <AlertTriangle className="h-4 w-4 shrink-0 text-red-600 dark:text-red-400" />
   419→      <div className="min-w-0 flex-1">
   420→        <div className="flex items-baseline gap-2 flex-wrap">
   421→          <span className="text-sm font-semibold truncate">
   422→            {announcement.title}
   423→          </span>
   424→          <span className="text-[10px] uppercase tracking-wide opacity-70 shrink-0">
   425→            Urgent · {relativeTime(announcement.createdAt)} · {announcement.author.name}
   426→          </span>
   427→        </div>
   428→        <p className="text-xs opacity-90 truncate hidden sm:block">{bodyPreview}</p>
   429→      </div>
   430→      <Button
   431→        type="button"
   432→        variant="club"
   433→        size="sm"
   434→        className="shrink-0 h-7"
   435→        onClick={() => setView("announcements")}
   436→      >
   437→        View
   438→      </Button>
   439→      <button
   440→        type="button"
   441→        onClick={dismiss}
   442→        aria-label="Dismiss urgent banner"
   443→        className="shrink-0 inline-flex h-7 w-7 items-center justify-center rounded-md hover:bg-red-100 dark:hover:bg-red-900/40 transition-colors"
   444→      >
   445→        <X className="h-3.5 w-3.5" />
   446→      </button>
   447→    </div>
   448→  )
   449→}

   952→/**
   953→ * Bell dropdown — recent notifications grouped by New / Earlier, each with a
   954→ * type-tinted icon, relative time, and click-to-navigate. Footer links open
   955→ * the full NotificationsView or the Notifications settings tab.

   957→ * Live updates: subscribes to `realtime:notification` events for instant
   958→ * refetch + toast on arrival; falls back to a 20s poll.
   959→ */
   960→function NotificationBell() {
   961→  const [items, setItems] = useState<NotifItem[]>([])
   962→  const [unread, setUnread] = useState(0)
   963→  const [open, setOpen] = useState(false)
   964→  const [markingAll, setMarkingAll] = useState(false)
   965→  const setView = useAppStore((s) => s.setView)
   966→
   967→  const refresh = useCallback(async () => {
   968→    try {
   969→      const data = await api<NotifListResponse>("/api/notifications?pageSize=10")
   970→      setItems(data.items)
   971→      setUnread(data.unread)
   972→    } catch {
   973→      /* noop — silent failure for the bell preview */
   974→    }
   975→  }, [])
   976→
   977→  useEffect(() => {
   978→    refresh()
   979→    const t = setInterval(refresh, 20_000)
   980→    // Instant refresh when a realtime notification arrives + toast.
   981→    const unsub = onRealtimeEvent("realtime:notification", (data: any) => {
   982→      refresh()
   983→      if (data?.message) {
   984→        toast.message(data.message, { description: "Just now" })
   985→      }
   986→    })
   987→    return () => { clearInterval(t); unsub() }
   988→  }, [refresh])
   989→
   990→  async function markAllRead() {
   991→    if (markingAll) return
   992→    setMarkingAll(true)
   993→    // Optimistic — flip everything immediately for snappy UX.
   994→    setItems((prev) => prev.map((i) => ({ ...i, isRead: true })))
   995→    setUnread(0)
   996→    try {
   997→      await api("/api/notifications/read-all", { method: "POST" })
   998→    } catch {
   999→      // Re-fetch on failure so we don't lie about state.
  1000→      refresh()
  1001→    } finally {
  1002→      setMarkingAll(false)
  1003→    }
  1004→  }
  1005→
  1006→  async function markRead(id: string) {
  1007→    // Optimistic — only flip the targeted row.
  1008→    setItems((prev) =>
  1009→      prev.map((i) => (i.id === id ? { ...i, isRead: true } : i))
  1010→    )
  1011→    setUnread((u) => Math.max(0, u - 1))
  1012→    try {
  1013→      await api(`/api/notifications/${id}/read`, { method: "POST" })
  1014→    } catch {
  1015→      refresh()
  1016→    }
  1017→  }
