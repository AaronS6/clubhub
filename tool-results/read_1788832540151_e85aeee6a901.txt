   351→/**
   352→ * Dismissible banner shown at the very top of the app whenever the current
   353→ * club has an active urgent announcement. Dismissal is stored in localStorage
   354→ * per-announcement (`urgent-dismissed-<id>`) and auto-expires after 7 days.
   355→ *
   356→ * Only the LATEST urgent announcement is shown — older urgents effectively
   357→ * retire themselves by being superseded (the API returns only the most
   358→ * recent one), so a new urgent clears the dismissed state by virtue of
   359→ * having a different ID.
   360→ */
   361→const URGENT_DISMISS_TTL_MS = 7 * 24 * 60 * 60 * 1000 // 7 days
   362→
   363→function UrgentBanner() {
   364→  const currentClubId = useAppStore((s) => s.currentClubId)
   365→  const setView = useAppStore((s) => s.setView)
   366→  const [dismissedId, setDismissedId] = useState<string | null>(null)
   367→
   368→  const { data } = useQuery<{ announcement: {
   369→    id: string
   370→    title: string
   371→    body: string
   372→    createdAt: string
   373→    author: { id: string; name: string }
   374→  } | null }>({
   375→    queryKey: ["urgent-announcement", currentClubId ?? ""],
   376→    queryFn: () => api(`/api/clubs/${currentClubId}/announcements/urgent`),
   377→    enabled: !!currentClubId,
   378→    // Poll every 60s — urgent banners are high-signal; we want a near-real-time
   379→    // surfacing of new urgent announcements without spamming the server.
   380→    refetchInterval: 60_000,
   381→    staleTime: 30_000,
   382→  })
   383→
   384→  const announcement = data?.announcement ?? null
   385→
   386→  // Read dismissal state from localStorage when the announcement changes.
   387→  // Deferred to a microtask so we don't call setState synchronously inside
   388→  // the effect body (which would trigger cascading renders per React 19's
   389→  // react-hooks/set-state-in-effect rule).
   390→  useEffect(() => {
   391→    let cancelled = false
   392→    Promise.resolve().then(() => {
   393→      if (cancelled) return
   394→      if (!announcement) {
   395→        setDismissedId(null)
   396→        return
   397→      }
   398→      try {
   399→        const key = `urgent-dismissed-${announcement.id}`
   400→        const raw = localStorage.getItem(key)
   401→        if (!raw) {
   402→          setDismissedId(null)
   403→          return
   404→        }
   405→        const parsed = JSON.parse(raw) as { expiresAt?: number }
   406→        if (parsed.expiresAt && parsed.expiresAt > Date.now()) {
   407→          setDismissedId(announcement.id)
   408→        } else {
   409→          // expired — clear it so the banner re-shows
   410→          localStorage.removeItem(key)
   411→          setDismissedId(null)
   412→        }
   413→      } catch {
   414→        setDismissedId(null)
   415→      }
   416→    })
   417→    return () => {
   418→      cancelled = true
   419→    }
   420→  }, [announcement])
   421→
   422→  // Realtime refresh: when an announcement is created/updated/deleted, the
   423→  // urgent endpoint should be re-fetched so the banner appears/disappears
   424→  // immediately rather than waiting for the next poll.
   425→  const qc = useQueryClient()
   426→  useEffect(() => {
   427→    const unsub = onRealtimeEvent("realtime:club", (d: any) => {
   428→      const t = d?.type as string | undefined
   429→      if (
   430→        t === "announcement_created" ||
   431→        t === "announcement_updated" ||
   432→        t === "announcement_deleted" ||
   433→        t === "new_announcement"
   434→      ) {
   435→        qc.invalidateQueries({ queryKey: ["urgent-announcement", currentClubId ?? ""] })
   436→      }
   437→    })
   438→    return () => unsub()
   439→  }, [currentClubId, qc])
   440→
   441→  function dismiss() {
   442→    if (!announcement) return
   443→    const expiresAt = Date.now() + URGENT_DISMISS_TTL_MS
   444→    try {
   445→      localStorage.setItem(
   446→        `urgent-dismissed-${announcement.id}`,
   447→        JSON.stringify({ expiresAt }),
   448→      )
   449→    } catch {
   450→      // ignore — localStorage may be unavailable in private browsing
   451→    }
   452→    setDismissedId(announcement.id)
   453→  }
   454→
   455→  if (!announcement || dismissedId === announcement.id) return null
   456→
   457→  const bodyPreview = announcement.body.length > 120
   458→    ? announcement.body.slice(0, 117) + "…"
   459→    : announcement.body
   460→
   461→  return (
   462→    <div
   463→      role="alert"
   464→      className="flex items-center gap-3 px-3 sm:px-4 py-2 border-b border-red-200 dark:border-red-900/60 bg-red-50 dark:bg-red-950/30 text-red-900 dark:text-red-100 animate-fade-in"
   465→    >
   466→      <AlertTriangle className="h-4 w-4 shrink-0 text-red-600 dark:text-red-400" />
   467→      <div className="min-w-0 flex-1">
   468→        <div className="flex items-baseline gap-2 flex-wrap">
   469→          <span className="text-sm font-semibold truncate">
   470→            {announcement.title}
   471→          </span>
   472→          <span className="text-[10px] uppercase tracking-wide opacity-70 shrink-0">
   473→            Urgent · {relativeTime(announcement.createdAt)} · {announcement.author.name}
   474→          </span>
   475→        </div>
   476→        <p className="text-xs opacity-90 truncate hidden sm:block">{bodyPreview}</p>
   477→      </div>
   478→      <Button
   479→        type="button"
   480→        variant="club"
   481→        size="sm"
   482→        className="shrink-0 h-7"
   483→        onClick={() => setView("announcements")}
   484→      >
   485→        View
   486→      </Button>
   487→      <button
   488→        type="button"
   489→        onClick={dismiss}
   490→        aria-label="Dismiss urgent banner"
   491→        className="shrink-0 inline-flex h-7 w-7 items-center justify-center rounded-md hover:bg-red-100 dark:hover:bg-red-900/40 transition-colors"
   492→      >
   493→        <X className="h-3.5 w-3.5" />
   494→      </button>
   495→    </div>
   496→  )
   497→}

   506→}) {

  1000→/**
  1001→ * Bell dropdown — recent notifications grouped by New / Earlier, each with a
  1002→ * type-tinted icon, relative time, and click-to-navigate. Footer links open
  1003→ * the full NotificationsView or the Notifications settings tab.
  1004→ *
  1005→ * Live updates: subscribes to `realtime:notification` events for instant
  1006→ * refetch + toast on arrival; falls back to a 20s poll.
  1007→ */
  1008→function NotificationBell() {
  1009→  const [items, setItems] = useState<NotifItem[]>([])
  1010→  const [unread, setUnread] = useState(0)
  1011→  const [open, setOpen] = useState(false)
  1012→  const [markingAll, setMarkingAll] = useState(false)
  1013→  const setView = useAppStore((s) => s.setView)
  1014→
  1015→  const refresh = useCallback(async () => {
  1016→    try {
  1017→      const data = await api<NotifListResponse>("/api/notifications?pageSize=10")
  1018→      setItems(data.items)
  1019→      setUnread(data.unread)
  1020→    } catch {
  1021→      /* noop — silent failure for the bell preview */
  1022→    }
  1023→  }, [])
  1024→
  1025→  useEffect(() => {
  1026→    refresh()
  1027→    const t = setInterval(refresh, 20_000)

  1064→    }

  1295→/**
  1296→ * Keyboard shortcuts help dialog.
  1297→ *
  1298→ * Listens globally for `?` (Shift+/) and opens a small Dialog listing every
  1299→ * keyboard shortcut available in the app. The listener is intentionally
  1300→ * suppressed when the user is typing in an input/textarea/contentEditable
  1301→ * element, or when a meta key (Cmd/Ctrl/Alt) is held — those combinations
  1302→ * belong to the OS or browser, not us.
  1303→ *
  1304→ * Other UI surfaces (the footer "Shortcuts" button, the top-bar `?` button)
  1305→ * dispatch an `open-keyboard-shortcuts` CustomEvent to trigger the same dialog
  1306→ * — same pattern the SettingsDialog uses for its `open-settings` event.
  1307→ */
  1308→function KeyboardShortcutsHelp() {
  1309→  const [open, setOpen] = useState(false)
  1310→
  1311→  useEffect(() => {
  1312→    function onKey(e: KeyboardEvent) {
  1313→      // Ignore when a meta/ctrl/alt key is held — those are browser/OS
  1314→      // shortcuts, not ours.
  1315→      if (e.metaKey || e.ctrlKey || e.altKey) return
  1316→      // Ignore when the user is typing in an input, textarea, or any
  1317→      // contentEditable element. (Selects are included too — typing `?`
  1318→      // inside a select wouldn't make sense anyway.)
  1319→      const t = e.target as HTMLElement | null
  1320→      if (t) {
  1321→        const tag = t.tagName
  1322→        if (
  1323→          tag === "INPUT" ||
  1324→          tag === "TEXTAREA" ||
  1325→          tag === "SELECT" ||
  1326→          t.isContentEditable
  1327→        ) {
  1328→          return
  1329→        }
  1330→      }
  1331→      if (e.key === "?") {
  1332→        e.preventDefault()
  1333→        setOpen(true)
  1334→      }
  1335→    }
  1336→    function onOpenEvent(e: Event) {
  1337→      setOpen(true)
  1338→    }
  1339→    window.addEventListener("keydown", onKey)
  1340→    window.addEventListener("open-keyboard-shortcuts", onOpenEvent as EventListener)
  1341→    return () => {
  1342→      window.removeEventListener("keydown", onKey)
  1343→      window.removeEventListener("open-keyboard-shortcuts", onOpenEvent as EventListener)
  1344→    }
  1345→  }, [])
  1346→
  1347→  return (
  1348→    <Dialog open={open} onOpenChange={setOpen}>
  1349→      <DialogContent className="max-w-md">
  1350→        <DialogHeader>
  1351→          <DialogTitle className="flex items-center gap-2">
  1352→            <Keyboard className="h-4 w-4 text-club" />
  1353→            Keyboard shortcuts
  1354→          </DialogTitle>
  1355→          <DialogDescription>
  1356→            Press these keys anywhere in the app to navigate faster.
  1357→          </DialogDescription>
  1358→        </DialogHeader>
  1359→        <ul className="space-y-2 mt-2">
  1360→          {SHORTCUTS.map((s) => (
  1361→            <li
  1362→              key={s.label}
  1363→              className="flex items-center justify-between gap-3 rounded-md border bg-card/40 px-3 py-2"
  1364→            >
  1365→              <span className="text-body-medium">{s.label}</span>
  1366→              <span className="flex items-center gap-1 shrink-0">
  1367→                {s.keys.map((k, i) => (
  1368→                  <kbd
  1369→                    key={i}
  1370→                    className="inline-flex items-center justify-center min-w-6 h-5 px-1.5 rounded border bg-muted text-[11px] font-mono text-foreground"
  1371→                  >
  1372→                    {k}
  1373→                  </kbd>
  1374→                ))}
  1375→              </span>
  1376→            </li>
  1377→          ))}
  1378→        </ul>
  1379→        <p className="text-caption text-muted-foreground mt-3">
  1380→          Tip: shortcuts are ignored while you&apos;re typing in a text field.
  1381→        </p>
  1382→      </DialogContent>
  1383→    </Dialog>
  1384→  )
  1385→}
