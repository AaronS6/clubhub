   293→/**
   294→ * Dismissible banner shown at the very top of the app whenever the current
   295→ * club has an active urgent announcement. Dismissal is stored in localStorage
   296→ * per-announcement (`urgent-dismissed-<id>`) and auto-expires after 7 days.
   297→ *
   298→ * Only the LATEST urgent announcement is shown — older urgents effectively
   299→ * retire themselves by being superseded (the API returns only the most
   300→ * recent one), so a new urgent clears the dismissed state by virtue of
   301→ * having a different ID.
   302→ */
   303→const URGENT_DISMISS_TTL_MS = 7 * 24 * 60 * 60 * 1000 // 7 days
   304→
   305→function UrgentBanner() {
   306→  const currentClubId = useAppStore((s) => s.currentClubId)
   307→  const setView = useAppStore((s) => s.setView)
   308→  const [dismissedId, setDismissedId] = useState<string | null>(null)
   309→
   310→  const { data } = useQuery<{ announcement: {
   311→    id: string
   312→    title: string
   313→    body: string
   314→    createdAt: string
   315→    author: { id: string; name: string }
   316→  } | null }>({
   317→    queryKey: ["urgent-announcement", currentClubId ?? ""],
   318→    queryFn: () => api(`/api/clubs/${currentClubId}/announcements/urgent`),
   319→    enabled: !!currentClubId,
   320→    // Poll every 60s — urgent banners are high-signal; we want a near-real-time
   321→    // surfacing of new urgent announcements without spamming the server.
   322→    refetchInterval: 60_000,
   323→    staleTime: 30_000,
   324→  })
   325→
   326→  const announcement = data?.announcement ?? null
   327→
   328→  // Read dismissal state from localStorage when the announcement changes.
   329→  // Deferred to a microtask so we don't call setState synchronously inside
   330→  // the effect body (which would trigger cascading renders per React 19's
   331→  // react-hooks/set-state-in-effect rule).
   332→  useEffect(() => {
   333→    let cancelled = false
   334→    Promise.resolve().then(() => {
   335→      if (cancelled) return
   336→      if (!announcement) {
   337→        setDismissedId(null)
   338→        return
   339→      }
   340→      try {
   341→        const key = `urgent-dismissed-${announcement.id}`
   342→        const raw = localStorage.getItem(key)
   343→        if (!raw) {
   344→          setDismissedId(null)
   345→          return
   346→        }
   347→        const parsed = JSON.parse(raw) as { expiresAt?: number }
   348→        if (parsed.expiresAt && parsed.expiresAt > Date.now()) {
   349→          setDismissedId(announcement.id)
   350→        } else {
   351→          // expired — clear it so the banner re-shows
   352→          localStorage.removeItem(key)
   353→          setDismissedId(null)
   354→        }
   355→      } catch {
   356→        setDismissedId(null)
   357→      }
   358→    })
   359→    return () => {
   360→      cancelled = true
   361→    }
   362→  }, [announcement])
   363→
   364→  // Realtime refresh: when an announcement is created/updated/deleted, the
   365→  // urgent endpoint should be re-fetched so the banner appears/disappears
   366→  // immediately rather than waiting for the next poll.
   367→  const qc = useQueryClient()
   368→  useEffect(() => {
   369→    const unsub = onRealtimeEvent("realtime:club", (d: any) => {
   370→      const t = d?.type as string | undefined
   371→      if (
   372→        t === "announcement_created" ||
   373→        t === "announcement_updated" ||
   374→        t === "announcement_deleted" ||
   375→        t === "new_announcement"
   376→      ) {
   377→        qc.invalidateQueries({ queryKey: ["urgent-announcement", currentClubId ?? ""] })
   378→      }
   379→    })
   380→    return () => unsub()
   381→  }, [currentClubId, qc])
   382→
   383→  function dismiss() {
   384→    if (!announcement) return
   385→    const expiresAt = Date.now() + URGENT_DISMISS_TTL_MS
   386→    try {
   387→      localStorage.setItem(
   388→        `urgent-dismissed-${announcement.id}`,
   389→        JSON.stringify({ expiresAt }),
   390→      )
   391→    } catch {
   392→      // ignore — localStorage may be unavailable in private browsing
   393→    }
   394→    setDismissedId(announcement.id)
   395→  }
   396→
   397→  if (!announcement || dismissedId === announcement.id) return null
   398→
   399→  const bodyPreview = announcement.body.length > 120
   400→    ? announcement.body.slice(0, 117) + "…"
   401→    : announcement.body
   402→
   403→  return (
   404→    <div
   405→      role="alert"
   406→      className="flex items-center gap-3 px-3 sm:px-4 py-2 border-b border-red-200 dark:border-red-900/60 bg-red-50 dark:bg-red-950/30 text-red-900 dark:text-red-100 animate-fade-in"
   407→    >
   408→      <AlertTriangle className="h-4 w-4 shrink-0 text-red-600 dark:text-red-400" />
   409→      <div className="min-w-0 flex-1">
   410→        <div className="flex items-baseline gap-2 flex-wrap">
   411→          <span className="text-sm font-semibold truncate">
   412→            {announcement.title}
   413→          </span>
   414→          <span className="text-[10px] uppercase tracking-wide opacity-70 shrink-0">
   415→            Urgent · {relativeTime(announcement.createdAt)} · {announcement.author.name}
   416→          </span>
   417→        </div>
   418→        <p className="text-xs opacity-90 truncate hidden sm:block">{bodyPreview}</p>
   419→      </div>
   420→      <Button
   421→        type="button"
   422→        variant="club"
   423→        size="sm"
   424→        className="shrink-0 h-7"
   425→        onClick={() => setView("announcements")}
   426→      >
   427→        View
   428→      </Button>
   429→      <button
   430→        type="button"
   431→        onClick={dismiss}
   432→        aria-label="Dismiss urgent banner"
   433→        className="shrink-0 inline-flex h-7 w-7 items-center justify-center rounded-md hover:bg-red-100 dark:hover:bg-red-900/40 transition-colors"
   434→      >
   435→        <X className="h-3.5 w-3.5" />
   436→      </button>
   437→    </div>
   438→  )
   439→}

   942→/**

   944→ * type-tinted icon, relative time, and click-to-navigate. Footer links open
   945→ * the full NotificationsView or the Notifications settings tab.
   946→ *
   947→ * Live updates: subscribes to `realtime:notification` events for instant
   948→ * refetch + toast on arrival; falls back to a 20s poll.
   949→ */
   950→function NotificationBell() {
   951→  const [items, setItems] = useState<NotifItem[]>([])
   952→  const [unread, setUnread] = useState(0)
   953→  const [open, setOpen] = useState(false)
   954→  const [markingAll, setMarkingAll] = useState(false)
   955→  const setView = useAppStore((s) => s.setView)
   956→
   957→  const refresh = useCallback(async () => {
   958→    try {
   959→      const data = await api<NotifListResponse>("/api/notifications?pageSize=10")
   960→      setItems(data.items)
   961→      setUnread(data.unread)
   962→    } catch {
   963→      /* noop — silent failure for the bell preview */
   964→    }
   965→  }, [])
   966→
   967→  useEffect(() => {
   968→    refresh()
   969→    const t = setInterval(refresh, 20_000)
   970→    // Instant refresh when a realtime notification arrives + toast.
   971→    const unsub = onRealtimeEvent("realtime:notification", (data: any) => {
   972→      refresh()
   973→      if (data?.message) {
   974→        toast.message(data.message, { description: "Just now" })
   975→      }
   976→    })
   977→    return () => { clearInterval(t); unsub() }
   978→  }, [refresh])
   979→
   980→  async function markAllRead() {
   981→    if (markingAll) return
   982→    setMarkingAll(true)
   983→    // Optimistic — flip everything immediately for snappy UX.
   984→    setItems((prev) => prev.map((i) => ({ ...i, isRead: true })))
   985→    setUnread(0)
   986→    try {
   987→      await api("/api/notifications/read-all", { method: "POST" })
   988→    } catch {
   989→      // Re-fetch on failure so we don't lie about state.
   990→      refresh()
   991→    } finally {
   992→      setMarkingAll(false)
   993→    }
   994→  }
   995→
   996→  async function markRead(id: string) {
   997→    // Optimistic — only flip the targeted row.
   998→    setItems((prev) =>
   999→      prev.map((i) => (i.id === id ? { ...i, isRead: true } : i))
  1000→    )
  1001→    setUnread((u) => Math.max(0, u - 1))
  1002→    try {
  1003→      await api(`/api/notifications/${id}/read`, { method: "POST" })
  1004→    } catch {
  1005→      refresh()
  1006→    }
  1007→  }
