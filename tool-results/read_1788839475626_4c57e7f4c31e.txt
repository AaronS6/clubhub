   142→/** Format a meeting's date/time: "Mon, Jan 15 · 3:00 PM – 4:30 PM" */
   143→function formatMeetingRange(start: string, end: string) {
   144→  const s = parseISO(start)
   145→  const e = parseISO(end)
   146→  return `${format(s, "EEE, MMM d")} · ${format(s, "h:mm a")} – ${format(e, "h:mm a")}`
   147→}
   148→
   149→/** Convert an ISO/date to a value usable by <input type="datetime-local"> */
   150→function toLocalInputValue(d: Date) {
   151→  // Build YYYY-MM-DDTHH:MM in local time
   152→  const pad = (n: number) => (n < 10 ? `0${n}` : `${n}`)
   153→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
   154→}

   156→function icsLinkFor(clubId: string) {
   157→  return `/api/clubs/${clubId}/meetings/ics`
   158→}

   196→    () => (teamFilter === "all" ? upcoming : upcoming.filter((m) => m.teamId === teamFilter)),

   380→function MeetingsList({
   381→  meetings,
   382→  isLoading,
   383→  isError,
   384→  isExec,
   385→  readOnly,
   386→  onRetry,
   387→  onEdit,
   388→  onOpenDetail,
   389→  emptyTitle,
   390→  emptyDescription,
   391→  emptyAction,
   392→}: {
   393→  meetings: MeetingItem[]
   394→  isLoading: boolean
   395→  isError: boolean
   396→  isExec: boolean
   397→  readOnly?: boolean
   398→  onRetry: () => void
   399→  onEdit: (m: MeetingItem) => void
   400→  onOpenDetail: (m: MeetingItem) => void
   401→  emptyTitle: string
   402→  emptyDescription?: string
   403→  emptyAction?: React.ReactNode
   404→}) {
   405→  if (isLoading) {
   406→    return (
   407→      <div className="grid gap-3">
   408→        {Array.from({ length: 3 }).map((_, i) => (
   409→          <Skeleton key={i} className="h-32 w-full rounded-xl" />
   410→        ))}
   411→      </div>
   412→    )
   413→  }
   414→  if (isError) {
   415→    return (
   416→      <EmptyState
   417→        icon={<X className="size-8" />}
   418→        title="Couldn't load meetings"
   419→        description="Something went wrong. Please try again."
   420→        action={
   421→          <Button variant="outline" size="sm" onClick={onRetry} className="gap-2">
   422→            <RefreshCw className="size-4" /> Retry
   423→          </Button>
   424→        }
   425→      />
   426→    )
   427→  }
   428→  if (meetings.length === 0) {
   429→    return (
   430→      <EmptyState
   431→        icon={<CalendarDays className="size-10" />}
   432→        title={emptyTitle}
   433→        description={emptyDescription}
   434→        action={emptyAction}
   435→      />
   436→    )
   437→  }
   438→  return (
   439→    <div className="grid gap-3">
   440→      {meetings.map((m) => (
   441→        <MeetingCard
   442→          key={m.id}
   443→          meeting={m}
   444→          isExec={isExec}
   445→          readOnly={readOnly}
   446→          onEdit={() => onEdit(m)}
   447→          onOpenDetail={() => onOpenDetail(m)}
   448→        />
   449→      ))}
   450→    </div>
   451→  )
   452→}

   454→function MeetingCard({
   455→  meeting,
   456→  isExec,
   457→  readOnly,
   458→  onEdit,
   459→  onOpenDetail,
   460→}: {
   461→  meeting: MeetingItem
   462→  isExec: boolean
   463→  readOnly?: boolean
   464→  onEdit: () => void
   465→  onOpenDetail: () => void
   466→}) {

   469→  // someone RSVP'd, so the change is perceptible across sessions.

   472→    <div className={cn(
   473→      "card-quiet overflow-hidden p-4 sm:p-5 animate-fade-in",
   474→      flash && "ring-2 ring-club/50 shadow-md"
   475→    )}>
   476→      <div className="flex flex-col gap-4 sm:flex-row sm:items-start sm:justify-between">
   477→        <div className="min-w-0 flex-1">

   479→            onClick={onOpenDetail}

   482→            <div className="min-w-0 flex-1">
   483→              <div className="flex flex-wrap items-center gap-2">
   484→                <h3 className="truncate text-card-title group-hover:text-club transition-colors">
   485→                  {meeting.title}
   486→                </h3>
   487→                {meeting.isRecurring && meeting.recurrenceRule && (
   488→                  <Badge variant="secondary" className="gap-1 capitalize">
   489→                    <RefreshCw className="size-3" />
   490→                    {meeting.recurrenceRule}
   491→                  </Badge>
   492→                )}
   493→                {meeting.team && (
   494→                  <Badge variant="outline" className="gap-1">
   495→                    <Users className="size-3" />
   496→                    {meeting.team.name}
   497→                  </Badge>
   498→                )}
   499→              </div>
   500→              <div className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-caption text-muted-foreground">
   501→                <span className="inline-flex items-center gap-1">
   502→                  <Clock className="size-3.5" />
   503→                  {formatMeetingRange(meeting.startTime, meeting.endTime)}
   504→                </span>
   505→                <span className="inline-flex items-center gap-1 min-w-0">
   506→                  <MapPin className="size-3.5 shrink-0" />
   507→                  <span className="max-w-[14rem] truncate">{meeting.location}</span>
   508→                </span>
   509→              </div>
   510→            </div>
   511→          </button>
   512→        </div>
   513→
   514→        <div className="flex shrink-0 items-center gap-2">
   515→          {isExec && !readOnly && (
   516→            <DropdownMenu>
   517→              <DropdownMenuTrigger asChild>
   518→                <Button variant="ghost" size="icon" aria-label="Meeting actions">
   519→                  <MoreVertical className="size-4" />
   520→                </Button>
   521→              </DropdownMenuTrigger>
   522→              <DropdownMenuContent align="end">
   523→                <DropdownMenuItem onClick={onEdit} className="gap-2">
   524→                  <Pencil className="size-4" /> Edit
   525→                </DropdownMenuItem>
   526→                <DropdownMenuSeparator />
   527→                <CancelMeetingMenuItem meeting={meeting} />
   528→              </DropdownMenuContent>
   529→            </DropdownMenu>
   530→          )}
   531→        </div>
   532→      </div>
   533→
   534→      <div className="mt-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   535→        <div className="text-caption text-muted-foreground">
   536→          {meeting.rsvpCounts.going} going · {meeting.rsvpCounts.maybe} maybe
   537→          {meeting.rsvpCounts.notGoing > 0 && ` · ${meeting.rsvpCounts.notGoing} not going`}
   538→        </div>
   539→        {!readOnly && (
   540→          <div className="flex flex-wrap gap-2">
   541→            <RsvpButton meeting={meeting} status="going" />

   543→            <RsvpButton meeting={meeting} status="not_going" />
   544→          </div>
   545→        )}
   546→        {readOnly && (
   547→          <div className="text-caption text-muted-foreground">
   548→            {format(start, "MMM d, yyyy")}
   549→          </div>
   550→        )}
   551→      </div>
   552→    </div>
   553→  )
   554→}

   556→function RsvpButton({ meeting, status }: { meeting: MeetingItem; status: RsvpStatus }) {
   557→  const queryClient = useQueryClient()
   558→  const clubId = useAppStore((s) => s.currentClubId)
   559→  const isCurrent = meeting.myRsvp === status
   560→
   561→  const mutation = useMutation({
   562→    mutationFn: () =>
   563→      api(`/api/clubs/${clubId}/meetings/${meeting.id}/rsvp`, {
   564→        method: "POST",
   565→        json: { status },
   566→      }),
   567→    onSuccess: () => {
   568→      queryClient.invalidateQueries({ queryKey: ["meetings", clubId] })
   569→    },
   570→    onError: (err: unknown) => {
   571→      toast.error(err instanceof Error ? err.message : "Failed to update RSVP")
   572→    },
   573→  })
   574→
   575→  const labels: Record<RsvpStatus, string> = {
   576→    going: "Going",
   577→    maybe: "Maybe",
   578→    not_going: "Not Going",
   579→  }
   580→  const icons: Record<RsvpStatus, React.ReactNode> = {
   581→    going: <Check className="size-3.5" />,
   582→    maybe: <HelpCircle className="size-3.5" />,
   583→    not_going: <X className="size-3.5" />,
   584→  }
   585→
   586→  return (
   587→    <Button
   588→      size="sm"
   589→      variant={isCurrent ? "club" : "outline"}
   590→      disabled={mutation.isPending}
   591→      onClick={() => mutation.mutate()}
   592→      className="gap-1.5"
   593→      aria-pressed={isCurrent}
   594→    >
   595→      {mutation.isPending && isCurrent ? <Loader2 className="size-3.5 animate-spin" /> : icons[status]}
   596→      <span className="hidden xs:inline sm:inline">{labels[status]}</span>
   597→    </Button>
   598→  )
   599→}
