   135→function linkify(text: string) {
   136→  // Combined URL + @mention renderer. URLs become clickable links; @mentions
   137→  // get a styled span (text-club font-medium) for visual emphasis. Mention
   138→  // click-through to a member profile is a future enhancement — for v1 we
   139→  // just style them.
   140→  const parts: React.ReactNode[] = []
   141→  const urlSplit = text.split(URL_REGEX)
   142→  urlSplit.forEach((chunk, i) => {
   143→    if (/^https?:\/\//.test(chunk)) {
   144→      parts.push(
   145→        <a
   146→          key={`u${i}`}
   147→          href={chunk}
   148→          target="_blank"
   149→          rel="noopener noreferrer"
   150→          className="text-club hover:underline break-all"
   151→        >
   152→          {chunk}
   153→        </a>,
   154→      )
   155→    } else {
   156→      let lastIdx = 0
   157→      const re = new RegExp(MENTION_REGEX)
   158→      let m: RegExpExecArray | null
   159→      while ((m = re.exec(chunk)) !== null) {
   160→        if (m.index > lastIdx) {
   161→          parts.push(<span key={`t${i}-${lastIdx}`}>{chunk.slice(lastIdx, m.index)}</span>)
   162→        }
   163→        parts.push(
   164→          <span
   165→            key={`m${i}-${m.index}`}
   166→            className="text-club font-medium"
   167→          >
   168→            {m[0]}
   169→          </span>,
   170→        )
   171→        lastIdx = m.index + m[0].length
   172→      }
   173→      if (lastIdx < chunk.length) {
   174→        parts.push(<span key={`t${i}-end`}>{chunk.slice(lastIdx)}</span>)
   175→      }
   176→    }
   177→  })
   178→  return parts
   179→}

   181→export function AnnouncementsView() {
   182→  const clubId = useAppStore((s) => s.currentClubId)
   183→  const currentClub = useAppStore((s) => s.currentClub)
   184→  const isExec = currentClub?.role === "executive"
   185→  const { data: session } = useSession()
   186→  const currentUserId = session?.user?.id
   187→  const currentUser = session?.user
   188→    ? {
   189→        id: session.user.id,
   190→        name: session.user.name,
   191→        avatarUrl: session.user.image ?? null,
   192→      }
   193→    : undefined
   194→
   195→  const [composeOpen, setComposeOpen] = useState(false)
   196→
   197→  const query = useInfiniteQuery<AnnouncementsResponse>({
   198→    queryKey: clubId ? [...announcementListKey(clubId)] : ["announcements", "none"],
   199→    queryFn: ({ pageParam }) =>
   200→      api<AnnouncementsResponse>(`/api/clubs/${clubId}/announcements?page=${pageParam}`),
   201→    initialPageParam: 1,
   202→    getNextPageParam: (lastPage) => (lastPage.hasMore ? lastPage.page + 1 : undefined),
   203→    enabled: !!clubId,
   204→    // Realtime is primary; poll only as a fallback while the socket is down
   205→    // (was previously always-on 10s polling — now degrades to polling only
   206→    // when the realtime connection is lost).
   207→    refetchInterval: usePollingFallback(10000),
   208→  })
   209→
   210→  const items: AnnouncementItem[] = query.data?.pages.flatMap((p) => p.items) ?? []
   211→  const hasMore = query.data?.pages[query.data.pages.length - 1]?.hasMore ?? false
   212→
   213→  function invalidateAll() {
   214→    if (!clubId) return
   215→    queryClient.invalidateQueries({ queryKey: announcementListKey(clubId) })
   216→  }
   217→
   218→  const queryClient = useQueryClient()
   219→
   220→  if (!clubId) {
   221→    return (
   222→      <div className="p-8 text-muted-foreground">Select a club to view announcements.</div>
   223→    )
   224→  }
   225→
   226→  return (
   227→    <div className="space-y-6">
   228→      <PageHeader
   229→        title="Announcements"
   230→        description="Stay up to date with club news, pinned notices, and discussions."
   231→        actions={
   232→          isExec ? (
   233→            <Button variant="club" onClick={() => setComposeOpen(true)}>
   234→              <Plus className="h-4 w-4" /> New Announcement
   235→            </Button>
   236→          ) : undefined
   237→        }
   238→      />
   239→
   240→      {query.isLoading ? (
   241→        <FeedSkeleton />
   242→      ) : items.length === 0 ? (
   243→        <EmptyState
   244→          icon={<Megaphone className="h-8 w-8" />}
   245→          title="No announcements yet"
   246→          description={
   247→            isExec
   248→              ? "Post your first announcement to keep members in the loop."
   249→              : "Check back later for updates from club executives."
   250→          }
   251→          action={
   252→            isExec ? (
   253→              <Button variant="club" onClick={() => setComposeOpen(true)}>
   254→                <Plus className="h-4 w-4" /> Post announcement
   255→              </Button>
   256→            ) : undefined
   257→          }
   258→        />
   259→      ) : (
   260→        <div className="space-y-4">
   261→          {items.map((a) => (
   262→            <AnnouncementCard
   263→              key={a.id}
   264→              announcement={a}
   265→              clubId={clubId}
   266→              isExec={isExec}
   267→              currentUserId={currentUserId}
   268→              currentUser={currentUser}
   269→              onMutated={invalidateAll}
   270→            />
   271→          ))}
   272→          {hasMore ? (
   273→            <div className="flex justify-center pt-2">
   274→              <Button
   275→                variant="outline"
   276→                onClick={() => query.fetchNextPage()}
   277→                disabled={query.isFetchingNextPage}
   278→              >
   279→                {query.isFetchingNextPage ? (
   280→                  <>
   281→                    <Loader2 className="h-4 w-4 animate-spin" /> Loading…
   282→                  </>
   283→                ) : (
   284→                  <>
   285→                    <ChevronDown className="h-4 w-4" /> Load more
   286→                  </>
   287→                )}
   288→              </Button>
   289→            </div>
   290→          ) : null}
   291→        </div>
   292→      )}
   293→
   294→      <ComposeAnnouncementDialog
   295→        open={composeOpen}
   296→        onOpenChange={setComposeOpen}
   297→        clubId={clubId}
   298→        onCreated={() => {
   299→          setComposeOpen(false)
   300→          invalidateAll()
   301→        }}
   302→      />
   303→    </div>
   304→  )
   305→}

   307→function AnnouncementCard({
   308→  announcement,
   309→  clubId,
   310→  isExec,
   311→  currentUserId,
   312→  currentUser,
   313→  onMutated,
   314→}: {
   315→  announcement: AnnouncementItem
   316→  clubId: string
   317→  isExec: boolean
   318→  currentUserId?: string
   319→  currentUser?: { id: string; name: string; avatarUrl: string | null }
   320→  onMutated: () => void
   321→}) {
   322→  const queryClient = useQueryClient()
   323→  const [commentsOpen, setCommentsOpen] = useState(false)
   324→  const [editing, setEditing] = useState(false)
   325→
   326→  const isAuthor = !!currentUserId && announcement.authorId === currentUserId
   327→  const canEdit = isExec || isAuthor
   328→
   329→  const reactionMutation = useMutation({
   330→    mutationFn: (emoji: string) =>
   331→      api<{ reactions: ReactionSummary[]; myReaction: string | null }>(
   332→        `/api/clubs/${clubId}/announcements/${announcement.id}/reactions`,
   333→        { method: "POST", json: { emoji } }
   334→      ),
   335→    onMutate: async (emoji) => {
   336→      // Optimistic update across all cached pages
   337→      await cancelAllAnnouncementQueries(queryClient, clubId)
   338→      const previous = queryClient.getQueriesData<{ pages: AnnouncementsResponse[] }>({
   339→        queryKey: [...announcementListKey(clubId)],
   340→      })
   341→      for (const [key, value] of previous) {
   342→        if (!value?.pages) continue
   343→        queryClient.setQueryData(key, {
   344→          ...value,
   345→          pages: value.pages.map((p) => ({
   346→            ...p,
   347→            items: p.items.map((it) =>
   348→              it.id === announcement.id ? applyReactionOptimistic(it, emoji, currentUser) : it
   349→            ),
   350→          })),
   351→        })
   352→      }
   353→      return { previous }
   354→    },
   355→    onError: (_err, _emoji, ctx) => {
   356→      if (ctx?.previous) {
   357→        for (const [key, value] of ctx.previous) {
   358→          queryClient.setQueryData(key, value)
   359→        }
   360→      }
   361→      toast.error("Could not update reaction")
   362→    },
   363→    onSuccess: (data) => {
   364→      patchAnnouncementInCache(queryClient, clubId, announcement.id, (it) => ({
   365→        ...it,
   366→        reactions: data.reactions,
   367→        myReaction: data.myReaction,
   368→      }))
   369→    },
   370→  })
   371→
   372→  const pinMutation = useMutation({
   373→    mutationFn: (isPinned: boolean) =>
   374→      api<{ item: AnnouncementItem }>(
   375→        `/api/clubs/${clubId}/announcements/${announcement.id}`,
   376→        { method: "PATCH", json: { isPinned } }
   377→      ),
   378→    onSuccess: (res) => {
   379→      patchAnnouncementInCache(queryClient, clubId, announcement.id, () => res.item)
   380→      toast.success(res.item.isPinned ? "Pinned to top" : "Unpinned")
   381→    },
   382→    onError: (e: Error) => toast.error(e.message),
   383→  })
   384→
   385→  const deleteMutation = useMutation({
   386→    mutationFn: () =>
   387→      api(`/api/clubs/${clubId}/announcements/${announcement.id}`, { method: "DELETE" }),
   388→    onSuccess: () => {
   389→      toast.success("Announcement deleted")
   390→      onMutated()
   391→    },
   392→    onError: (e: Error) => toast.error(e.message),
   393→  })
   394→
   395→  return (
   396→    <div className="card-quiet p-0 gap-0 overflow-hidden animate-fade-in">
   397→      <div className="p-4 md:p-5 space-y-3">
   398→        {/* Header */}
   399→        <div className="flex items-start justify-between gap-3">
   400→          <div className="flex items-center gap-3 min-w-0">
   401→            <Avatar className="h-10 w-10">
   402→              <AvatarImage src={announcement.author.avatarUrl ?? undefined} alt={announcement.author.name} />
   403→              <AvatarFallback>{initials(announcement.author.name)}</AvatarFallback>
   404→            </Avatar>
   405→            <div className="min-w-0">
   406→              <div className="flex items-center gap-2 flex-wrap">
   407→                <span className="text-body-medium truncate">{announcement.author.name}</span>
   408→                {announcement.isUrgent && (
   409→                  <span className="inline-flex items-center gap-1 rounded-full bg-red-100 text-red-800 dark:bg-red-950 dark:text-red-200 px-2 py-0.5 text-[10px] font-semibold">
   410→                    <AlertTriangle className="h-3 w-3" /> Urgent
   411→                  </span>
   412→                )}
   413→                {announcement.isPinned && (
   414→                  <span className="inline-flex items-center gap-1 rounded-full bg-amber-100 text-amber-800 dark:bg-amber-950 dark:text-amber-200 px-2 py-0.5 text-[10px] font-semibold">
   415→                    <Pin className="h-3 w-3" /> Pinned
   416→                  </span>
   417→                )}
   418→              </div>
   419→              <div className="text-caption">
   420→                {relativeTime(announcement.createdAt)}
   421→                {new Date(announcement.updatedAt).getTime() - new Date(announcement.createdAt).getTime() > 1000 && (
   422→                  <span className="ml-1 italic">(edited)</span>
   423→                )}
   424→              </div>
   425→            </div>
   426→          </div>
   427→          {(canEdit || isExec) && (
   428→            <AnnouncementMenu
   429→              isExec={isExec}
   430→              canEdit={canEdit}
   431→              pinned={announcement.isPinned}
   432→              onTogglePin={() => pinMutation.mutate(!announcement.isPinned)}
   433→              onEdit={() => setEditing(true)}
   434→              onDelete={() => deleteMutation.mutate()}
   435→              pinLoading={pinMutation.isPending}
   436→              deleteLoading={deleteMutation.isPending}
   437→            />
   438→          )}
   439→        </div>
   440→
   441→        {/* Title + Body */}
   442→        {announcement.title && (
   443→          <h3 className="text-card-title leading-tight">{announcement.title}</h3>
   444→        )}
   445→        <div className="text-body whitespace-pre-wrap break-words leading-relaxed">
   446→          {linkify(announcement.body)}
   447→        </div>
   448→
   449→        {/* Reaction bar */}
   450→        <ReactionBar
   451→          announcement={announcement}
   452→          myReaction={announcement.myReaction}
   453→          currentUserId={currentUserId}
   454→          onReact={(emoji) => reactionMutation.mutate(emoji)}
   455→          disabled={reactionMutation.isPending}
   456→        />
   457→      </div>
   458→
   459→      {/* Footer: comment toggle */}
   460→      <div className="border-t bg-muted/20 px-4 md:px-5 py-2">
   461→        <button
   462→          type="button"
   463→          onClick={() => setCommentsOpen((v) => !v)}
   464→          className="inline-flex items-center gap-1.5 text-caption-medium text-muted-foreground hover:text-foreground transition-colors"
   465→          aria-expanded={commentsOpen}
   466→        >
   467→          <MessageSquare className="h-3.5 w-3.5" />
   468→          {announcement.commentCount} {announcement.commentCount === 1 ? "comment" : "comments"}
   469→          <ChevronDown className={cn("h-3 w-3 transition-transform", commentsOpen && "rotate-180")} />
   470→        </button>
   471→      </div>
   472→
   473→      {commentsOpen && (
   474→        <CommentSection
   475→          clubId={clubId}
   476→          announcementId={announcement.id}
   477→          isExec={isExec}
   478→          onCommentCountChanged={() => onMutated()}
   479→        />
   480→      )}
   481→
   482→      {editing && (
   483→        <EditAnnouncementDialog
   484→          open={editing}
   485→          onOpenChange={setEditing}
   486→          clubId={clubId}
   487→          announcement={announcement}
   488→          isExec={isExec}
   489→          onSaved={() => {
   490→            setEditing(false)
   491→            onMutated()
   492→          }}
   493→        />
   494→      )}
   495→    </div>
   496→  )
   497→}

   565→function ReactionBar({
   566→  announcement,
   567→  myReaction,
   568→  currentUserId,
   569→  onReact,
   570→  disabled,
   571→}: {
   572→  announcement: AnnouncementItem
   573→  myReaction: string | null
   574→  currentUserId?: string
   575→  onReact: (emoji: string) => void
   576→  disabled?: boolean
   577→}) {
   578→  const activeReactions = announcement.reactions // only entries with count > 0 are returned by the API
   579→  return (
   580→    <div className="flex items-center flex-wrap gap-1.5 pt-1">
   581→      {/* Existing reaction badges — only render ones with at least 1 reaction */}
   582→      {activeReactions.map((entry) => {
   583→        const isMine = myReaction === entry.emoji
   584→        const users = entry.users
   585→        return (
   586→          <Tooltip key={entry.emoji}>
   587→            <TooltipTrigger asChild>
   588→              <button
   589→                type="button"
   590→                onClick={() => onReact(entry.emoji)}
   591→                disabled={disabled}
   592→                aria-pressed={isMine}
   593→                aria-label={`React ${entry.emoji} · ${entry.count} ${entry.count === 1 ? "person" : "people"}`}
   594→                className={cn(
   595→                  "inline-flex items-center gap-1 rounded-full border px-2 py-1 text-xs transition-all h-7",
   596→                  isMine
   597→                    ? "bg-club-muted border-transparent text-club ring-1 ring-club"
   598→                    : "bg-background border-border hover:bg-muted text-foreground hover:border-foreground/20"
   599→                )}
   600→              >
   601→                <span aria-hidden>{entry.emoji}</span>
   602→                <span className="font-medium tabular-nums">{entry.count}</span>
   603→                {isMine && (
   604→                  <span className="sr-only">(your reaction — click to remove)</span>
   605→                )}
   606→              </button>
   607→            </TooltipTrigger>
   608→            <TooltipContent side="top" className="max-w-[240px]">
   609→              <div className="text-xs">
   610→                <div className="font-semibold mb-1">
